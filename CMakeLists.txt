cmake_minimum_required(VERSION 3.25)
project(AutoGDBot VERSION 0.5.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(AUTOGDBOT_BUILD_MOD "Build the Geode mod" ON)
option(AUTOGDBOT_BUILD_TESTS "Build standalone controller tests" OFF)
if(AUTOGDBOT_BUILD_TESTS)
 enable_testing()
 add_executable(controller_tests tests/controller_tests.cpp)
 target_include_directories(controller_tests PRIVATE src)
 add_test(NAME controller_tests COMMAND controller_tests)
endif()
if(AUTOGDBOT_BUILD_MOD)
 if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Set GEODE_SDK to the Geode 5.10.1 SDK directory")
 endif()
 if(ANDROID)
  add_library(${PROJECT_NAME} SHARED src/main.cpp src/gpu_ranker_stub.cpp)
 else()
  add_library(${PROJECT_NAME} SHARED src/main.cpp src/gpu_ranker.cpp)
 endif()
 target_include_directories(${PROJECT_NAME} PRIVATE src)
 add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
 if(WIN32)
  target_link_libraries(${PROJECT_NAME} d3d11 d3dcompiler dxgi)
 endif()
 setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
endif()
