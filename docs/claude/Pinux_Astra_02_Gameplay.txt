  private static final class DrawItem {
    String art;
    float x, y, artX, artY;
    int dir, frame, outfit;
    boolean person;

    DrawItem(String art, float x, float y, float ax, float ay) {
      this.art = art;

      this.y = y;

  @Override
  public void create() {
    content = Content.load();
    saves = new SaveStore(SaveStore.defaultDirectory(), content);
    settings = Settings.load(saves.directory());
    c = new PixelCanvas();
    c.preload();
    life = new LifeScreens(this, c);
    audio = new AudioScene();
    world = new World(content, new GameState());
    rebuildDrawItems();
    snapCamera();
    windowScale = settings.scale;
    selection = saves.exists() ? 0 : 1;
    if (!Boolean.getBoolean("ctrl.test")) applyDisplay();
    Gdx.input.setInputProcessor(
        new InputMultiplexer(life.pcNativeStage(), new InputAdapter() {
          @Override
          public boolean keyDown(int key) {
            return press(key);
          }

          @Override
          public boolean keyUp(int key) {
            held.remove(key);
            if (key == Keys.TAB) showDetails = false;
            return true;
          }

          @Override
          public boolean keyTyped(char ch) {
            if (mode == Mode.CREATE) {
              if (ch == '\b' && !name.isEmpty()) name = name.substring(0, name.length() - 1);
              else if (!Character.isISOControl(ch)
                  && (Character.isLetterOrDigit(ch) || ch == ' ' || ch == '-')
                  && name.length() < 18) name += ch;
            } else if (mode == Mode.IDE) {
              life.pcKeyTyped(ch);
            }
            return true;
          }

          @Override
          public boolean touchDown(int x, int y, int pointer, int button) {
            if (button == Input.Buttons.LEFT) {
              int px = Math.round(c.logicalX(x)), py = Math.round(c.logicalY(y));
              if (mode == Mode.IDE && life.pcPointerDown(px, py)) {
                clicked = false;
                return true;
              }
              clicked = true;
              clickX = c.logicalX(x);
              clickY = c.logicalY(y);
            }
            return true;
          }

          @Override
          public boolean touchDragged(int x, int y, int pointer) {
            if (mode == Mode.IDE)
              return life.pcPointerDragged(Math.round(c.logicalX(x)), Math.round(c.logicalY(y)));
            return false;
          }

          @Override
          public boolean touchUp(int x, int y, int pointer, int button) {
            if (mode == Mode.IDE && button == Input.Buttons.LEFT)
              return life.pcPointerUp(Math.round(c.logicalX(x)), Math.round(c.logicalY(y)));
            return false;
          }

          @Override
          public boolean mouseMoved(int x, int y) {
            mouseMoved = true;
            if (mode == Mode.IDE) life.pcPointerMove(Math.round(c.logicalX(x)), Math.round(c.logicalY(y)));
            return true;
          }
        }));
  }

  public void uiHotspot(String label, int x, int y, int w, int h, Runnable action) {

    buttons.add(b);
    if (mouseMoved && contains(b, c.mouseX(), c.mouseY())) selection = index;
  }

  private void change(Mode next) {
    if (mode == Mode.IDE && next != Mode.IDE && life != null) life.pcCancelPointer();
    mode = next;
    held.clear();
    accumulator = 0;
    selection = next == Mode.MENU && !saves.exists() ? 1 : 0;
    confirm = false;
    clicked = false;
    showDetails = false;
    buttons.clear();
    world.moving = false;
  }

  private boolean press(int key) {
    if (bindIndex >= 0) {
      if (key == Keys.ESCAPE) {
        bindIndex = -1;
        return true;
      }
      if (settings.bind(bindIndex, key)) {
        bindIndex = -1;
        persistSettings();
      } else notify("Эта клавиша занята. Выбери другую.");
      return true;
    }
    if (key == Keys.F11) {
      settings.fullscreen = !settings.fullscreen;
      applyDisplay();
      persistSettings();
      held.clear();
      return true;
    }
    if (mode == Mode.IDE && life.pcKeyDown(key)) return true;
    if (mode == Mode.WORLD) {
      held.add(key);
      if (key == settings.interact || key == Keys.ENTER) {
        interact();
      } else if (key == Keys.ESCAPE) {
        change(Mode.PAUSE);
        saveQuiet();
      } else if (key == settings.journal) {
        change(Mode.JOURNAL);
      } else if (key == Keys.R) {
        change(Mode.CONTACTS);
      } else if (key == Keys.B) {
        change(Mode.BUDGET);
      } else if (key == Keys.M) {
        change(Mode.MAP);
      } else if (key == Keys.I) {
        change(Mode.BAG);
      } else if (key == Keys.F5) {
        save();
      } else if (key == Keys.TAB) {
        showDetails = true;
      }
      return true;
    }
    if (key == Keys.ESCAPE) {
      back();
      return true;
    }
    if (key == Keys.UP || key == Keys.DOWN) {
      int direction = key == Keys.UP ? -1 : 1;
      int next = selection + direction;
      while (next >= 0 && next < buttons.size() && !buttons.get(next).enabled) next += direction;
      if (next >= 0 && next < buttons.size()) selection = next;
      return true;
    }
    if (key == Keys.ENTER) {
      confirm = true;
      return true;
    }
    if (mode == Mode.DIALOGUE && key >= Keys.NUM_1 && key <= Keys.NUM_3) {
      selection = key - Keys.NUM_1;
      confirm = true;
    }
    return true;
  }

  private void back() {
    switch (mode) {
      case MENU -> {
        /* Exit is an explicit menu action. */
      }
      case CREATE, CONFIRM_NEW -> change(Mode.MENU);
      case SETTINGS -> {
        persistSettings();
        change(settingsReturn);
      }
      case CONTROLS -> change(Mode.SETTINGS);
      case INTRO -> {
        world.state.introSeen = true;
        change(Mode.WORLD);
        saveQuiet();
      }
      case IDE, SHOP, SOCIAL, CONTACTS, REST, SCENE, NIGHT, BUDGET, PAUSE, JOURNAL, MAP, BAG -> {
        change(Mode.WORLD);
        saveQuiet();
      }
      case DIALOGUE -> {
        change(Mode.WORLD);
        saveQuiet();
      }
      default -> change(Mode.PAUSE);
    }
  }

  private void openSettings(Mode from) {
    settingsReturn = from;
    change(Mode.SETTINGS);
  }

  private void newGame() {
    GameState state = new GameState();
    state.name = name.strip().isEmpty() ? "Саша" : name.strip();
    state.outfit = newOutfit;
    world = new World(content, state);
    hasSession = true;
    rebuildDrawItems();
    snapCamera();
    life.introPage = 0;
    change(Mode.INTRO);
    arrivalTimer = 4;
    saveQuiet();
    notify("WASD — идти · E — действие · J — журнал");
  }

  private void continueGame() {
    try {
      var loaded = saves.load();
      world = new World(content, loaded.state());
      hasSession = true;
      rebuildDrawItems();
      snapCamera();
      life.introPage = 0;
      change(world.state.introSeen ? Mode.WORLD : Mode.INTRO);
      arrivalTimer = 3;
      if (loaded.fromBackup()) notify("Восстановлена предыдущая резервная копия.");
    } catch (IOException e) {
      notify("Сохранение не прочитано. Можно начать новую игру.");
    }
  }

  private void save() {
    try {
      saves.save(world.state);
      notify("Игра сохранена.");
    } catch (IOException | RuntimeException e) {
      notify("Не удалось сохранить. Проверь доступ к папке данных.");
      Gdx.app.error("Save", "Не удалось сохранить игру", e);
    }
  }

  private void saveQuiet() {
    if (!hasSession) return;
    try {
      saves.save(world.state);
    } catch (IOException | RuntimeException e) {
      notify("Автосохранение не удалось. Нажми F5 для повтора.");
      Gdx.app.error("Save", "Автосохранение", e);
    }
  }

  private void persistSettings() {
    try {
      settings.save(saves.directory());
    } catch (IOException e) {
      notify("Не удалось сохранить настройки.");
    }
  }

  private void applyDisplay() {
    if (settings.fullscreen) Gdx.graphics.setFullscreenMode(Gdx.graphics.getDisplayMode());
    else {
      var display = Gdx.graphics.getDisplayMode();
      int max = Math.max(2, Math.min(display.width / 480, display.height / 270));
      windowScale = Math.min(settings.scale, max);
      Gdx.graphics.setWindowedMode(480 * windowScale, 270 * windowScale);
    }
  }

  public void focusChanged(boolean focused) {
    this.focused = focused;
    held.clear();
    accumulator = 0;
    showDetails = false;
    if (!focused && mode == Mode.WORLD) {
      change(Mode.PAUSE);
      saveQuiet();
    }
  }

  @Override
  public void pause() {
    focusChanged(false);
  }

  @Override
  public void resume() {
    focused = true;
    held.clear();
    accumulator = 0;
  }

  private void notify(String text) {
    toast = text;
    toastTimer = 4.5f;
  }

  private void interact() {
    var npc = world.interactionNpc();
    var prop = world.nearestProp();
    if (npc != null) {
      if (npc.id.equals("mira") || npc.id.equals("asya") || npc.id.equals("nika")) {
        if (npc.id.equals(world.state.plannedGirl)
            && world.state.map.equals(world.state.plannedVenue)) life.openDate();
        else life.openSocial(npc.id, false);
        return;
      }
      String reaction = Career.community(world.state, world.state.map);
      if (npc.id.equals("tamara") && !reaction.isEmpty()) {
        life.scene(
            "Тамара", reaction + "\n\nЕда — у стойки. Рада, что ты заглянул(а).", Mode.WORLD);
        return;
      }
      openDialogue(world.npcDialogue(npc));
      return;
    }
    if (prop == null) return;
    audio.click(settings);
    switch (prop.action) {
      case "mentor" -> {
        life.openCode();
        life.codePage = 8;
        return;
      }
      case "computer" -> {
        life.openCode();
        return;
      }
      case "shop" -> {
        life.shopPage = 0;
        change(Mode.SHOP);
        return;
      }
      case "rest" -> {
        change(Mode.REST);
        return;
      }
      case "date" -> {
        life.openDate();
        return;
      }
      case "courier" -> {
        life.scene("Подработка без ПК", Life.courier(world.state), Mode.WORLD);
        return;
      }
      case "wait" -> {
        Life.advance(world.state, 30);
        world.state.energy = Math.min(100, world.state.energy + 8);
        life.scene("Пауза в сквере", "Прошло 30 минут. Ты немного отдохнул(а).", Mode.WORLD);
        return;
      }
    }
    if (prop.action.equals("door")) {
      audio.door(settings);
      doorFade = settings.reducedEffects ? 0 : .18f;
      world.enter(prop);
      rebuildDrawItems();
      snapCamera();
      held.clear();
      accumulator = 0;
      arrivalTimer = 3;
      saveQuiet();
    } else if (prop.action.equals("wardrobe")) {
      world.state.outfit = (world.state.outfit + 1) % 3;
      notify("Куртка: " + new String[] {"медовая", "дождливая", "кирпичная"}[world.state.outfit]);
      saveQuiet();
    } else openDialogue(world.propDialogue(prop));
  }

  private void openDialogue(String id) {
    dialogue = content.dialogue(id);
    change(Mode.DIALOGUE);
    audio.click(settings);
  }

  private void choose(Content.Choice choice) {
    boolean delivered = world.state.deliveredLetter;
    world.effect(choice.effect);
    rebuildDrawItems();
    saveQuiet();
    if (!delivered && world.state.deliveredLetter) {
      audio.success(settings);
      notify("Письмо у Миры. В журнале новая запись.");
    }
    if (choice.next.isEmpty()) change(Mode.WORLD);
    else {
      dialogue = content.dialogue(choice.next);
      selection = 0;
    }
  }

  private float targetCameraX() {
    return Math.max(0, Math.min(world.map().width - 480, world.state.x - 240));
  }

  private float targetCameraY() {
    return Math.max(0, Math.min(world.map().height - 270, world.state.y - 172));
  }

  private void snapCamera() {
    cameraX = targetCameraX();
    cameraY = targetCameraY();
  }

  private void rebuildDrawItems() {
    drawItems.clear();
    for (var p : world.map().objects)
      if (p.art != null && world.visible(p)) {
        String art = p.art;
        if (p.id.equals("boxes") && world.state.unpacked) art = "boxes_open";
        if (p.id.equals("plant") && world.state.wateredPlant) art = "plant_watered";
        if (p.id.equals("post") && world.state.signedMailbox) art = "mailbox_signed";
        drawItems.add(
            new DrawItem(
                art,
                p.x,
                p.solid != null ? p.solid[1] + p.solid[3] : p.artY + c.artHeight(art),
                p.artX,
                p.artY));
      }
    for (var n : world.npcs()) {
      DrawItem d = new DrawItem(n.art, 0, 0, 0, 0);
      d.person = true;
      drawItems.add(d);
    }
    DrawItem hero = new DrawItem("hero", world.state.x, world.state.y, 0, 0);
    hero.person = true;
    drawItems.add(hero);
  }

  @Override
  public void render() {
    float delta = Math.min(Gdx.graphics.getDeltaTime(), .1f);
    visualTime += delta;
    toastTimer = Math.max(0, toastTimer - delta);
    arrivalTimer = Math.max(0, arrivalTimer - delta);
    doorFade = Math.max(0, doorFade - delta);
    if (mode == Mode.WORLD && focused) {
      accumulator += delta;
      float dx =
          (held.contains(settings.right) || held.contains(Keys.RIGHT) ? 1 : 0)
              - (held.contains(settings.left) || held.contains(Keys.LEFT) ? 1 : 0);
      float dy =
          (held.contains(settings.down) || held.contains(Keys.DOWN) ? 1 : 0)
              - (held.contains(settings.up) || held.contains(Keys.UP) ? 1 : 0);
      while (accumulator >= World.STEP) {
        world.tick(dx, dy);
        accumulator -= World.STEP;
      }
      if (world.moving) {
        footTimer += delta;
        if (footTimer > .30f) {
          audio.step(settings);
          footTimer = 0;
        }
      }
      float blend = 1 - (float) Math.exp(-delta * 9);
      cameraX += (targetCameraX() - cameraX) * blend;
      cameraY += (targetCameraY() - cameraY) * blend;
    } else accumulator = 0;
    audio.update(
        settings,
        (world.state.map.equals("street") || world.state.map.equals("park"))
            && Atmosphere.rain(world.state),
        focused);
    buttons.clear();
    c.begin();
    drawWorld();
    if (mode == Mode.WORLD) drawGuide();
    if (doorFade > 0) {
      transitionColor.a = doorFade / .18f;
      c.rect(0, 0, 480, 270, transitionColor);
    }
    switch (mode) {
      case MENU -> drawMenu();
      case CREATE -> drawCreate();
      case CONFIRM_NEW -> drawConfirm();
      case WORLD -> drawHud();
      case DIALOGUE -> {
        drawHud();
        drawDialogue();
      }
      case PAUSE -> drawPause();
      case SETTINGS -> drawSettings();
      case CONTROLS -> drawControls();
      case JOURNAL, MAP, BAG, INTRO, SHOP, IDE, SOCIAL, CONTACTS, REST, SCENE, NIGHT, BUDGET ->
          life.draw(mode);
    }
    if (toastTimer > 0 && mode == Mode.WORLD) {
      int w = Math.min(460, (int) Math.ceil(c.width(toast)) + 18);
      int y = 57;
      c.rect((480 - w) / 2, y, w, 20, "182a35f5");
      c.fitted(toast, (480 - w) / 2f + 9, y + 4, w - 18, MINT, false);
    }
    c.end();
    if (mode == Mode.IDE) life.renderPcOverlay(delta);
    activateButtons();
    mouseMoved = false;
  }

  private void drawWorld() {
    float ox = Math.round(cameraX), oy = Math.round(cameraY);
    c.art(world.state.map, -ox, -oy);
    var boundary = SceneLayout.bounds(world.state.map);
    c.clip(boundary.x() - ox, boundary.y() - oy, boundary.width(), boundary.height());
    if (world.state.map.equals("street")) {
      String[] names = {"ОБЩЕЖИТИЕ № 7", "ПЛЁНКА", "МАСТЕРСКАЯ", "ЛОЖКА"};
      int[] left = {76, 300, 411, 628}, widths = {200, 86, 185, 217};
      for (int i = 0; i < names.length; i++) {
        c.rect(left[i] - ox, 145 - oy, widths[i], 12, "203c46");
        c.rect(left[i] - ox, 145 - oy, widths[i], .5f, "b7b397");
        c.fitted(
            names[i],
            left[i] + (widths[i] - c.smallWidth(names[i])) / 2 - ox,
            146 - oy,
            widths[i] - 4,
            "ead7ae",
            true);
      }
      c.rect(850 - ox, 415 - oy, 85, 14, "203c46");
      c.fitted("СКВЕР →", 862 - ox, 417 - oy, 69, "d6d9b5", true);
    }
    for (var item : drawItems)
      if (item.person) {
        if (item.art.equals("hero")) {
          item.x = world.state.x;
          item.y = world.state.y;
          item.dir = world.state.direction;
          item.frame = world.moving ? (int) (world.state.elapsed * 8) % 4 : 0;
          item.outfit = world.state.outfit;
        } else
          for (var npc : world.npcs())
            if (npc.art.equals(item.art)) {
              world.position(npc, actor);
              item.x = actor.x;
              item.y = actor.y;
              item.dir = actor.direction;
              item.frame = actor.moving ? (int) (world.state.elapsed * 5) % 4 : 0;
            }
      }
    drawItems.sort(depth);
    for (var item : drawItems) {
      if (item.person) {
        c.scaledArt("shadow", item.x - 9 - ox, item.y - 4 - oy, 18, 7);
        c.actor(item.art, item.dir, item.frame, item.x - ox, item.y - oy, item.outfit);
      } else {
        String drawnArt =
            item.art.equals("desk")
                    && world.state.map.equals("room")
                    && world.state.upgrades.contains("ssd")
                ? "desk_ssd"
                : item.art;
        if (item.art.equals("desk")
            && world.state.map.equals("room")
            && world.state.upgrades.contains("chair")) c.art("chair_upgrade", 204 - ox, 121 - oy);
        c.scaledArt("shadow", item.artX - 2 - ox, item.y - 7 - oy, c.artWidth(drawnArt) + 4, 13);
        if (item.artX + c.artWidth(drawnArt) > ox
            && item.artX < ox + 480
            && item.artY + c.artHeight(drawnArt) > oy
            && item.artY < oy + 270) c.art(drawnArt, item.artX - ox, item.artY - oy);
      }
    }
    float night = Atmosphere.night(world.state);
    boolean outside = world.state.map.equals("street") || world.state.map.equals("park");
    dusk.a = night * (outside ? .32f : .18f);
    c.rect(0, 0, 480, 270, dusk);
    if (world.state.map.equals("room")) {
      c.art("light_monitor", 107 - ox, 64 - oy);
      c.glow("light_window", 170 - ox, 55 - oy, .3f + night * .7f);
      c.rect(211 - ox, 121 - oy, 2, 2, ((int) (visualTime * 2) % 2 == 0) ? "c4c992" : "68756e");
      if (!settings.reducedEffects) {
        int lift = (int) (visualTime * 4) % 4;
        c.rect(163 - ox, 110 - oy - lift, 1, 2, "d8d3b14a");
      }
    }
    if (world.state.map.equals("stair"))
      c.glow("light_amber", 152 - ox, 40 - oy, .4f + night * .6f);
    if (world.state.map.equals("cafe") || world.state.map.equals("diner"))
      c.art("light_amber", 90 - ox, 68 - oy);
    if (world.state.map.equals("street")) {

package life.ctrl.bytescript;

import java.util.*;

/** A bounded expression language, never evaluated by Java, a shell or an external service. */
public final class ByteScript {
  public enum Kind {
    SYNTAX,
    SEMANTIC
  }

  public static final class Error extends RuntimeException {
    public final Kind kind;
    public final int line, column;

    public Error(Kind kind, String message, int line, int column) {
      super(message);
      this.kind = kind;
      this.line = line;
      this.column = column;
    }
  }

  public record Program(String name, List<String> parameters, Expr expression) {
    public int operations(double... args) {
      run(args);
      Map<String, Double> values = new HashMap<>();
      for (int i = 0; i < args.length; i++) values.put(parameters.get(i), args[i]);
      return count(expression, values);
    }

    private static int count(Expr e, Map<String, Double> v) {
      if (e instanceof Binary b) return 1 + count(b.left, v) + count(b.right, v);
      if (e instanceof Select t)
        return 1 + count(t.condition, v) + count(t.condition.eval(v) != 0 ? t.yes : t.no, v);
      return 0;
    }

    public double run(double... args) {
      if (args.length != parameters.size()) throw new IllegalArgumentException("Число аргументов");
      Map<String, Double> values = new HashMap<>();
      for (int i = 0; i < args.length; i++) values.put(parameters.get(i), args[i]);
      double result = expression.eval(values);
      if (!Double.isFinite(result))
        throw new ArithmeticException("Деление на ноль или слишком большое число");
      return result;
    }
  }

  public sealed interface Expr permits Number, Variable, Binary, Select {
    double eval(Map<String, Double> values);
  }

  public record Number(double value) implements Expr {
    public double eval(Map<String, Double> v) {
      return value;
    }
  }

  public record Variable(String name) implements Expr {
    public double eval(Map<String, Double> v) {
      return v.get(name);
    }
  }

  public record Binary(String op, Expr left, Expr right) implements Expr {
    public double eval(Map<String, Double> v) {
      double a = left.eval(v), b = right.eval(v);
      return switch (op) {
        case "+" -> a + b;
        case "-" -> a - b;
        case "*" -> a * b;
        case "/" -> a / b;
        case ">=" -> a >= b ? 1 : 0;
        case "<=" -> a <= b ? 1 : 0;
        case ">" -> a > b ? 1 : 0;
        case "<" -> a < b ? 1 : 0;
        case "==" -> a == b ? 1 : 0;
        default -> throw new IllegalStateException(op);
      };
    }
  }

  public record Select(Expr condition, Expr yes, Expr no) implements Expr {
    public double eval(Map<String, Double> v) {
      return condition.eval(v) != 0 ? yes.eval(v) : no.eval(v);
    }
  }

  private record Token(String text, int line, int column) {}

  private final List<Token> tokens = new ArrayList<>();
  private final Set<String> names = new HashSet<>();
  private int at;

  private ByteScript(String source) {
    if (source.length() > 4096) throw new Error(Kind.SYNTAX, "Программа слишком длинная", 1, 1);
    int line = 1, col = 1;
    for (int i = 0; i < source.length(); ) {
      char c = source.charAt(i);
      if (Character.isWhitespace(c)) {
        if (c == '\n') {
          line++;
          col = 1;
        } else col++;
        i++;
        continue;
      }
      int begin = i, startCol = col;
      if (Character.isLetter(c) || c == '_') {
        while (i < source.length()
            && (Character.isLetterOrDigit(source.charAt(i)) || source.charAt(i) == '_')) i++;
      } else if (Character.isDigit(c)) {
        while (i < source.length()
            && (Character.isDigit(source.charAt(i)) || source.charAt(i) == '.')) i++;
      } else if ("(){};,?:+-*/<>=".indexOf(c) >= 0) {
        i++;
        if (i < source.length() && source.charAt(i) == '=' && "<>= ".indexOf(c) >= 0) i++;
      } else throw new Error(Kind.SYNTAX, "Неожиданный символ «" + c + "»", line, col);
      String text = source.substring(begin, i);
      tokens.add(new Token(text, line, startCol));
      col += text.length();
    }
    if (tokens.size() > 512) throw new Error(Kind.SYNTAX, "Слишком много операций", line, col);
    tokens.add(new Token("<end>", line, col));
  }

  public static Program compile(String source) {
    return new ByteScript(source).parse();
  }

  private Program parse() {
    expect("fn");
    String name = identifier();
    expect("(");
    List<String> params = new ArrayList<>();
    if (!peek(")")) {
      do {
        String p = identifier();
        if (!names.add(p)) fail(Kind.SEMANTIC, "Повтор аргумента: " + p);
        params.add(p);
      } while (take(","));
    }
    expect(")");
    expect("{");
    expect("return");
    Expr expr = select(0);
    expect(";");
    expect("}");
    expect("<end>");
    return new Program(name, List.copyOf(params), expr);
  }

  private Expr select(int depth) {
    if (depth > 48) fail(Kind.SYNTAX, "Слишком глубокое выражение");
    Expr c = compare(depth + 1);
    if (take("?")) {
      Expr y = select(depth + 1);
      expect(":");
      return new Select(c, y, select(depth + 1));
    }
    return c;
  }

  private Expr compare(int d) {
    Expr a = add(d);
    while (Set.of(">=", "<=", ">", "<", "==").contains(current().text)) {
      String op = next().text;
      a = new Binary(op, a, add(d));
    }
    return a;
  }

  private Expr add(int d) {
    Expr a = multiply(d);
    while (peek("+") || peek("-")) {
      String op = next().text;
      a = new Binary(op, a, multiply(d));
    }
    return a;
  }

  private Expr multiply(int d) {
    Expr a = atom(d);
    while (peek("*") || peek("/")) {
      String op = next().text;
      a = new Binary(op, a, atom(d));
    }
    return a;
  }

  private Expr atom(int d) {
    if (d > 48) fail(Kind.SYNTAX, "Слишком глубокое выражение");
    if (take("(")) {
      Expr e = select(d + 1);
      expect(")");
      return e;
    }
    if (take("-")) return new Binary("-", new Number(0), atom(d + 1));
    Token token = next();
    try {
      if (Character.isDigit(token.text.charAt(0)))
        return new Number(Double.parseDouble(token.text));
    } catch (NumberFormatException e) {
      fail(Kind.SYNTAX, "Неверное число");
    }
    if (!token.text.matches("[A-Za-z_][A-Za-z0-9_]*"))
      throw new Error(Kind.SYNTAX, "Ожидалось число или аргумент", token.line, token.column);
    if (!names.contains(token.text))
      throw new Error(
          Kind.SEMANTIC,
          "Неизвестное имя «" + token.text + "». Сверь его с аргументами функции.",
          token.line,
          token.column);
    return new Variable(token.text);
  }

  private String identifier() {
    Token t = next();
    if (!t.text.matches("[A-Za-z_][A-Za-z0-9_]*"))
      throw new Error(Kind.SYNTAX, "Ожидалось имя", t.line, t.column);
    return t.text;
  }

  private Token current() {
    return tokens.get(Math.min(at, tokens.size() - 1));
  }

  private Token next() {
    Token t = current();
    at++;
    return t;
  }

  private boolean peek(String s) {
    return current().text.equals(s);
  }

  private boolean take(String s) {
    if (peek(s)) {
      at++;
      return true;
    }
    return false;
  }

  private void expect(String s) {
    if (!take(s)) fail(Kind.SYNTAX, "Ожидалось «" + s + "», получено «" + current().text + "»");
  }

  private void fail(Kind k, String message) {
    Token t = current();
    throw new Error(k, message, t.line, t.column);
  }
}

package life.ctrl.model;

/** Continuous dusk, dawn and deterministic weather; rendering never changes saved simulation. */
public final class Atmosphere {
  private Atmosphere() {}

  public static float night(GameState s) {
    double hour = (s.gameMinutes % 1440) / 60;
    if (hour < 5 || hour >= 22) return 1;
    if (hour < 8) return (float) ((8 - hour) / 3);
    if (hour < 17) return 0;
    return (float) ((hour - 17) / 5);
  }

  public static boolean rain(GameState s) {
    return Math.floorMod(s.day(), 3) != 2;
  }

  public static String period(GameState s) {
    double h = (s.gameMinutes % 1440) / 60;

  }

  public static String rank(GameState s) {

      case 2 -> "Разработчик";
      case 1 -> "Фрилансер";
      default -> "Первый коммит";
    };

  }

    int n = 0;
    for (String id : List.of("board", "remind", "support")) {
      if (!s.shipped.contains(id)) break;
      n++;
    }
    return n;
  }

  public static String blocked(GameState s, LifeContent.Job j) {
    if (j.once) {
      int stage = projectStage(s);
      int required = List.of("board", "remind", "support").indexOf(j.id);

      if (required != stage) return "Сначала заверши предыдущий этап своего бота.";
      if (s.completedOrders < 1) return "Сначала выполни первый заказ для клиента.";
    } else {
      if (paidCountToday(s) >= 3)
        return "Три клиентских заказа на сегодня выполнены. Время для своего проекта или прогулки.";
      if (Coding.paidToday(s, j.id)) return "Этот заказ уже оплачен сегодня.";
    }
    if (tier(s) < j.tier)
      return "Нужно опыта: " + (j.tier == 1 ? 3 : 8) + ". Сейчас " + s.codingXp + ".";
    return "";
  }

    List<LifeContent.Job> result = new ArrayList<>();
    for (var j : data.jobs) if (!j.once && j.tier == 0) result.add(j);
    List<LifeContent.Job> advanced =
        Arrays.stream(data.jobs).filter(j -> !j.once && j.tier > 0 && j.tier <= tier(s)).toList();
    if (!advanced.isEmpty())
      for (int i = 0; i < Math.min(3, advanced.size()); i++)
        result.add(advanced.get(Math.floorMod(s.day() + i - 1, advanced.size())));
    return result;
  }

import java.util.*;

public final class Coding {
  private Coding() {}

  public record Report(boolean compiled, boolean passed, String message, List<String> cases) {}

  public static final class Revision {

    public Revision() {}

    public Revision(String source, String label) {

      this.label = label;
    }
  }

  public static String ledgerId(GameState s, String job) {

  }

  public static boolean paidToday(GameState s, String job) {
    return s.paidJobs.contains(ledgerId(s, job));
  }

  public static String accept(GameState s, LifeContent.Job j, int algorithm) {

  }

  public static String accept(GameState s, LifeContent.Job j, int algorithm, boolean rush) {

  }

  /** Use tactile typing in the interactive IDE; direct model callers retain older semantics. */
  public static String accept(GameState s, LifeContent.Job j, int algorithm, boolean rush, boolean tactile) {
    if (!s.activeJob.isEmpty() && s.jobStage > 0 && s.jobStage < 5)
      return "Текущий проект сохранён. Продолжи его или отложи через историю версий.";
    String blocked = Career.blocked(s, j);
    if (!blocked.isEmpty()) return blocked;
    if (algorithm < 0 || algorithm >= j.initialSources.length) return "Выбери вариант из задания.";
    s.activeJob = j.id;
    s.activeLedger = ledgerId(s, j.id);
    s.algorithm = algorithm;
    s.codeSource = j.initialSources[algorithm];
    s.typedCount = tactile ? 0 : j.kind.equals("feature") ? 0 : s.codeSource.length();
    s.jobStage = 2;
    s.programPreviewed = false;
    s.rushJob = rush && !j.once;
    s.rushDeadline =
        s.rushJob
            ? s.rushPromises.computeIfAbsent(
                ledgerId(s, j.id), key -> s.minutes() + Career.duration(s, j) + 90)
            : -1;
    s.revisions.clear();
    s.customCases.clear();
    s.revisions.add(new Revision(s.codeSource, "Исходная версия · " + s.clock()));
    return "Проект открыт. Прочитай требование и проверь граничные случаи.";
  }

  public static void type(GameState s) {
    if (s.jobStage == 2) s.typedCount = Math.min(s.codeSource.length(), s.typedCount + 5);
  }

  public static void completeTyping(GameState s) {
    s.typedCount = s.codeSource.length();
  }

  public static Report test(GameState s, LifeContent.Job j) {
    if (!s.activeJob.equals(j.id))
      return new Report(false, false, "Этот заказ не открыт в редакторе.", List.of());
    if (s.typedCount < s.codeSource.length())
      return new Report(
          false, false, "Сначала допечатай исходник. Быстрые реакции не нужны.", List.of());
    try {
      var p = ByteScript.compile(s.codeSource);
      if (!p.name().equals(j.function))
        return new Report(true, false, "Клиент ожидает функцию " + j.function, List.of());
      List<String> results = new ArrayList<>();
      boolean passed = true;
      int maxOps = 0;
      for (var t : j.tests) {
        double actual = p.run(t.args);
        boolean ok = Math.abs(actual - t.expected) < .001;
        passed &= ok;
        maxOps = Math.max(maxOps, p.operations(t.args));
        results.add(
            Arrays.toString(t.args)
                + " → "
                + format(actual)
                + " / нужно "
                + format(t.expected)
                + (ok ? "  OK" : "  ОШИБКА"));
      }
      boolean fast = j.operationBudget == 0 || maxOps <= j.operationBudget;
      passed &= fast;
      s.jobStage = passed ? 3 : 2;
      return new Report(
          true,
          passed,
          !fast
              ? "Ответ верный, но операций "
                  + maxOps
                  + ". Бюджет: "
                  + j.operationBudget
                  + ". Открой профилировщик."
              : passed
                  ? "Все тесты пройдены. Проверь предпросмотр и отправляй."
                  : "Результат не совпал. Сравни вход, ответ и требование.",
          List.copyOf(results));
    } catch (ByteScript.Error e) {
      s.jobStage = 2;
      return new Report(
          false,
          false,
          (e.kind == ByteScript.Kind.SYNTAX ? "Синтаксис" : "Неизвестное имя")
              + ", строка "
              + e.line
              + ": "
              + e.getMessage(),
          List.of());
    } catch (ArithmeticException | IllegalArgumentException e) {
      s.jobStage = 2;
      return new Report(true, false, e.getMessage(), List.of());
    }
  }

  public static String format(double x) {
    return x == Math.rint(x)
        ? String.valueOf((long) x)
        : String.valueOf(Math.round(x * 100.0) / 100.0);
  }

  public static void repair(GameState s, LifeContent.Job j, int choice) {
    repair(s, j, choice, false);
  }

  public static void repair(GameState s, LifeContent.Job j, int choice, boolean tactile) {
    if (!s.activeJob.equals(j.id) || s.jobStage == 5 || choice < 0 || choice >= j.solutions.length)
      return;
    s.codeSource = j.solutions[choice];
    // The IDE requires a second typing phase after a patch is chosen.
    s.typedCount = tactile ? 0 : s.codeSource.length();
    s.jobStage = 2;
    s.programPreviewed = false;
  }

  public static String commit(GameState s) {
    if (s.jobStage < 2 || s.jobStage == 5) return "Сначала открой проект.";
    if (s.revisions.stream().anyMatch(r -> r.source.equals(s.codeSource)))
      return "Эта версия уже сохранена. Коммит фиксирует изменение, а не повторный клик.";
    s.revisions.add(new Revision(s.codeSource, "Коммит " + s.revisions.size() + " · " + s.clock()));
    if (s.revisions.size() > 8) s.revisions.remove(1);
    return "Коммит сохранён. Теперь можно экспериментировать и вернуться к нему.";
  }

  public static String restore(GameState s, int index) {
    if (index < 0 || index >= s.revisions.size() || s.jobStage == 5) return "Версия недоступна.";
    s.codeSource = s.revisions.get(index).source;
    s.typedCount = s.codeSource.length();
    s.jobStage = 2;
    s.programPreviewed = false;
    return "Версия восстановлена. Прогони тесты заново.";
  }

  public static String abandon(GameState s) {
    s.activeJob = "";
    s.activeLedger = "";
    s.codeSource = "";
    s.typedCount = 0;
    s.jobStage = 0;
    s.programPreviewed = false;
    s.revisions.clear();
    s.customCases.clear();
    s.rushJob = false;
    return "Заказ отложен без штрафа. Можно выбрать другой.";
  }

  public static String probe(GameState s, LifeContent.Job j, double[] inputs) {
    if (!s.activeJob.equals(j.id) || s.typedCount < s.codeSource.length())
      return "Сначала допечатай исходник.";
    if (inputs.length != j.seed.length
        || Arrays.stream(inputs).anyMatch(v -> !Double.isFinite(v) || Math.abs(v) > 10000))
      return "Ввод: конечные числа от −10000 до 10000.";
    try {
      var p = ByteScript.compile(s.codeSource);
      double actual = p.run(inputs), expected = ByteScript.compile(j.source).run(inputs);
      String key = Arrays.toString(inputs);
      boolean novel = Arrays.stream(j.tests).noneMatch(t -> Arrays.equals(t.args, inputs));
      if (novel && !s.customCases.contains(key) && s.customCases.size() < 20)
        s.customCases.add(key);
      return "Ответ "
          + format(actual)
          + " · нужно "
          + format(expected)
          + " · операций "
          + p.operations(inputs)
          + (Math.abs(actual - expected) < .001 ? " · OK" : " · ОШИБКА");
    } catch (RuntimeException e) {
      return "Запуск не удался: " + e.getMessage();
    }
  }

  public static double preview(GameState s, LifeContent.Job j, double input) {
    if (!test(s, j).passed) throw new IllegalStateException("Сначала исправь программу");
    double[] args = j.id.equals("total") ? new double[] {30, input} : j.seed.clone();
    args[args.length - 1] = input;
    s.programPreviewed = true;
    return ByteScript.compile(s.codeSource).run(args);
  }

  public static String deliver(GameState s, LifeContent.Job j) {
    if (!s.activeJob.equals(j.id) || !s.activeLedger.endsWith(":" + j.id))
      return "Сначала открой нужный заказ.";
    if (s.jobStage == 5
        || s.paidJobs.contains(s.activeLedger)
        || (j.once && s.shipped.contains(j.id))) return "Эта работа уже сдана.";
    if (!s.programPreviewed || !test(s, j).passed) return "Нужны пройденные тесты и предпросмотр.";
    if (!j.once && Career.paidCountToday(s) >= 3)
      return "Сегодня уже оплачены три заказа. Продолжи завтра; исходник сохранён.";
    if (s.money < j.cost)
      return "Для запуска нужно " + j.cost + " ₽. Код сохранён, можно вернуться после подработки.";
    int time = Career.duration(s, j), bonus = Career.bonus(s, j);
    boolean late = s.rushJob && s.minutes() + time > s.rushDeadline;
    Economy.spend(s, j.cost, "Свой бот: " + j.title);
    if (!j.once) {
      s.paidJobs.add(ledgerId(s, j.id));
      if (!s.activeLedger.equals(ledgerId(s, j.id))) s.paidJobs.add(s.activeLedger);
      Economy.earn(s, j.pay + bonus, "Заказ: " + j.title);
      s.completedOrders++;
    }
    if (!s.shipped.contains(j.id)) s.shipped.add(j.id);
    s.codingXp += j.xp;
    s.jobStage = 5;
    Life.advance(s, time);
    s.energy = Math.max(0, s.energy - (s.upgrades.contains("chair") ? 2 : 6));
    s.mood = Math.min(100, s.mood + 8);
    s.lastClientMessage = j.feedback;
    return (j.once
            ? "Этап запущен. Расход: " + j.cost + " ₽."
            : "Клиент принял работу: +"
                + (j.pay + bonus)
                + " ₽"
                + (bonus > 0 ? " (бонус " + bonus + " ₽)" : "")
                + ".")
        + " Прошло "
        + time
        + " мин. Опыт +"
        + j.xp
        + ".\n\n"
        + j.feedback
        + (late ? "\n\nСрочный бонус истёк; базовая оплата сохранена." : "");
  }

package life.ctrl.model;

public final class Economy {
  private Economy() {}

  public static final int WEEKLY_RENT = 3000;

  public static final class Entry {
    public int day, amount, balance;
    public String reason;

    public Entry() {}

    public Entry(GameState s, int amount, String reason) {
      day = s.day();
      this.amount = amount;
      balance = s.money;
      this.reason = reason;
    }
  }

  public static boolean spend(GameState s, int amount, String reason) {
    if (amount < 0) throw new IllegalArgumentException("Отрицательная цена");
    if (s.money < amount) return false;
    if (amount == 0) return true;
    s.money -= amount;
    record(s, -amount, reason);
    return true;
  }

  public static void earn(GameState s, int amount, String reason) {
    if (amount < 0 || amount > 1000000 || s.money > Integer.MAX_VALUE - amount)
      throw new IllegalArgumentException("Сумма");
    s.money += amount;
    record(s, amount, reason);
  }

  private static void record(GameState s, int amount, String reason) {
    s.transactions.add(new Entry(s, amount, reason));
    if (s.transactions.size() > 100) s.transactions.remove(0);
  }

  public static void rent(GameState s) {
    while (s.day() > s.rentThroughDay) {
      int paid = Math.min(WEEKLY_RENT, s.money);
      spend(s, paid, "Комната: неделя аренды");
      s.rentDebt += WEEKLY_RENT - paid;
      s.rentThroughDay += 7;
    }
  }

  public static String repay(GameState s) {
    int amount = Math.min(s.rentDebt, s.money);
    if (amount == 0) return "Сейчас нечего погашать.";
    spend(s, amount, "Погашение долга за комнату");
    s.rentDebt -= amount;
    return "Долг уменьшен на " + amount + " ₽. Осталось: " + s.rentDebt + " ₽.";
  }
}

package life.ctrl.model;

import life.ctrl.content.LifeContent;

/** Economy and needs have a single mutation point, including explicit time costs. */
public final class Life {
  private Life() {}

  public static void advance(GameState s, double minutes) {
    if (!Double.isFinite(minutes) || minutes < 0 || minutes > 1440)
      throw new IllegalArgumentException("Время");
    s.gameMinutes += minutes;
    s.hunger = Math.max(0, s.hunger - minutes * .065);
    s.energy = Math.max(0, s.energy - minutes * .045);
    if (s.hunger < 10) s.health = Math.max(15, s.health - minutes * .012);
    Career.settleProject(s);
    Economy.rent(s);
  }

  public static String buy(GameState s, LifeContent.Item item) {
    if (s.money < item.price)
      return "Не хватает денег. Подработка у доски во дворе доступна всегда.";
    Economy.spend(s, item.price, "Покупка: " + item.name);
    s.items.merge(item.id, 1, Integer::sum);
    return "Куплено: " + item.name + ". [I] — съесть или посмотреть подарки.";
  }

  public static String eat(GameState s, LifeContent.Item item) {
    if (!item.kind.equals("food") || s.count(item.id) < 1) return "Сначала купи еду у стойки.";
    consume(s, item.id);
    s.hunger = Math.min(100, s.hunger + item.hunger);
    s.energy = Math.min(100, s.energy + item.energy);

    int n = s.count(item);
    if (n < 1) throw new IllegalStateException("Нет предмета");
    if (n == 1) s.items.remove(item);
    else s.items.put(item, n - 1);
  }

  public static void sleep(GameState s) {
    int wait = (480 - s.minutes() % 1440 + 1440) % 1440;

  public static Goal localGoal(World w, Goal finalGoal) {
    if (w.state.map.equals(finalGoal.map)) return finalGoal;
    record Hop(String map, Content.Prop first) {}
    ArrayDeque<Hop> q = new ArrayDeque<>();
    Set<String> seen = new HashSet<>();
    q.add(new Hop(w.state.map, null));
    while (!q.isEmpty()) {
      var h = q.remove();
      if (!seen.add(h.map)) continue;
      for (var p : w.content.map(h.map).objects)
        if (p.action.equals("door")) {
          var first = h.first == null ? p : h.first;
          if (p.target.equals(finalGoal.map))
            return new Goal(
                w.state.map,
                first.x,
                first.y,
                finalGoal.title,
                "Следующий проход: " + w.content.map(first.target).name);
          q.add(new Hop(p.target, first));
        }
    }
    return finalGoal;
  }

  public static List<float[]> route(World w, float targetX, float targetY) {
    int grid = 4, cols = w.map().width / grid, rows = w.map().height / grid;
    int start = Math.round(w.state.y / grid) * cols + Math.round(w.state.x / grid);
    int[] previous = new int[cols * rows];
    Arrays.fill(previous, -1);
    if (start < 0 || start >= previous.length) return List.of();
    previous[start] = start;
    ArrayDeque<Integer> todo = new ArrayDeque<>();
    todo.add(start);
    int goal = -1;
    while (!todo.isEmpty()) {
      int at = todo.remove(), x = at % cols, y = at / cols;
      if (Math.hypot(x * grid - targetX, y * grid - targetY) < 17) {
        goal = at;
        break;
      }
      for (int[] d : DIRS) {
        int nx = x + d[0], ny = y + d[1];
        if (nx < 0 || ny < 0 || nx >= cols || ny >= rows) continue;
        int next = ny * cols + nx;
        if (previous[next] < 0
            && !w.blocked(nx * grid - 2, ny * grid - 2)
            && !w.blocked(nx * grid + 2, ny * grid - 2)
            && !w.blocked(nx * grid - 2, ny * grid + 2)
            && !w.blocked(nx * grid + 2, ny * grid + 2)) {
          previous[next] = at;
          todo.add(next);
        }
      }
    }
    if (goal < 0) return List.of();
    ArrayList<float[]> path = new ArrayList<>();
    for (int p = goal; p != start; p = previous[p])
      path.add(new float[] {p % cols * grid, p / cols * grid});
    Collections.reverse(path);
    return path;
  }

  private static final int[][] DIRS = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
}

      down = Keys.S,

  public int key(int index) {
    return switch (index) {
      case 0 -> up;
      case 1 -> down;
      case 2 -> left;
      case 3 -> right;
      case 4 -> interact;
      default -> journal;
    };
  }

  public boolean bind(int index, int key) {
    if (key == Keys.ESCAPE
        || key == Keys.ENTER
        || key == Keys.F11
        || key == Keys.F5
        || key == Keys.R
        || key == Keys.B
        || key == Keys.I
        || key == Keys.M
        || key == Keys.TAB
        || key == Keys.UNKNOWN
        || key == Keys.UP
        || key == Keys.DOWN
        || key == Keys.LEFT
        || key == Keys.RIGHT) return false;
    for (int i = 0; i < 6; i++) if (i != index && key(i) == key) return false;
    switch (index) {
      case 0 -> up = key;
      case 1 -> down = key;
      case 2 -> left = key;
      case 3 -> right = key;
      case 4 -> interact = key;
      case 5 -> journal = key;
      default -> throw new IllegalArgumentException();
    }
    return true;
  }

  public static Settings load(Path dir) {
    try {
      Settings s =
          new Json().fromJson(Settings.class, Files.readString(dir.resolve("settings.json")));
      if (s == null
          || !Float.isFinite(s.music)
          || !Float.isFinite(s.sfx)
          || s.music < 0
          || s.music > 1
          || s.sfx < 0
          || s.sfx > 1
          || s.scale < 2
          || s.scale > 4) throw new IOException();
      HashSet<Integer> keys = new HashSet<>();
      for (int i = 0; i < 6; i++)
        if (s.key(i) <= 0 || s.key(i) > 255 || !keys.add(s.key(i)) || !s.bind(i, s.key(i)))
          throw new IOException();
      return s;
    } catch (Exception e) {
      return new Settings();
    }
  }

    Files.createDirectories(dir);
    Path tmp = dir.resolve("settings.tmp.json");
    Files.writeString(tmp, new Json().prettyPrint(this));
    SaveStore.replace(tmp, dir.resolve("settings.json"));
  }

    if (!cacheMap.equals(state.map)
        || !cacheGirl.equals(state.plannedGirl)
        || !cacheVenue.equals(state.plannedVenue)) {

      for (var m : content.maps)

          if (meeting ? state.map.equals(state.plannedVenue) : m.id.equals(state.map))

        || state.lastClientMessage == null
        || state.workNotice == null) throw new IllegalArgumentException("Некорректная карьера");
    for (var rev : state.revisions)
      if (rev == null || rev.source == null || rev.source.length() > 4096 || rev.label == null)
        throw new IllegalArgumentException("Некорректная версия проекта");
    for (var item : state.items.entrySet())
      if (item.getKey() == null || item.getValue() == null || item.getValue() <= 0)
        throw new IllegalArgumentException("Некорректный инвентарь");
    for (var relation : state.relationships.values())
      if (relation == null
          || relation.affinity < 0
          || relation.affinity > 100
          || relation.trust < 0
          || relation.trust > 100
          || relation.dates < 0
          || relation.sharedNights < 0
          || relation.memory == null) throw new IllegalArgumentException("Некорректные отношения");
    if (blocked(state.x, state.y))
      throw new IllegalArgumentException("Персонаж внутри препятствия");
    if (state.hasLetter && state.deliveredLetter)
      throw new IllegalArgumentException("Противоречивое состояние письма");
  }

  private static boolean validNeed(double value) {
    return Double.isFinite(value) && value >= 0 && value <= 100;
  }

  public void tick(float dx, float dy) {
    state.elapsed += STEP;
    Life.advance(state, STEP * .5);
    Social.expire(state);
    float length = (float) Math.hypot(dx, dy);
    moving = length > 0;
    if (!moving) return;
    dx = dx / length * SPEED * STEP;
    dy = dy / length * SPEED * STEP;
    state.direction = Math.abs(dx) > Math.abs(dy) ? (dx < 0 ? 1 : 2) : (dy < 0 ? 3 : 0);
    if (!blocked(state.x + dx, state.y)) state.x += dx;
    if (!blocked(state.x, state.y + dy)) state.y += dy;
  }

  public boolean blocked(float x, float y) {
    var m = map();
    if (x < RADIUS || y < RADIUS || x > m.width - RADIUS || y > m.height - RADIUS) return true;
    for (var r : m.solids) if (intersects(x, y, r)) return true;
    for (var p : m.objects) if (p.solid != null && intersects(x, y, p.solid)) return true;
    return false;
  }

  public static boolean intersects(float x, float y, float[] r) {
    return x + RADIUS > r[0] && x - RADIUS < r[0] + r[2] && y + 2 > r[1] && y - 4 < r[1] + r[3];
  }

  public boolean visible(Content.Prop p) {
    return !p.id.equals("letter") || (!state.hasLetter && !state.deliveredLetter);
  }

  public Content.Prop nearestProp() {
    Content.Prop found = null;
    double best = 29;
    for (var p : map().objects) {
      if (p.action.equals("decoration") || !visible(p)) continue;
      double dist = Math.hypot(state.x - p.x, state.y - p.y);
      if (dist < best) {
        best = dist;
        found = p;
      }
    }
    return found;
  }

  public static final class Actor {
    public float x, y;
    public int direction;
    public boolean moving;
  }

  public void position(Content.Npc n, Actor out) {
    if (n.id.equals(state.plannedGirl) && state.map.equals(state.plannedVenue)) {
      out.x = state.map.equals("park") ? 264 : 328;
      out.y = state.map.equals("park") ? 155 : 216;
      out.direction = 1;
      out.moving = false;
      return;
    }
    // A loop with a four-second pause at every waypoint; derived from saved world time.
    double duration = 0;
    int count = n.route.length;
    for (int i = 0; i < count; i++)
      duration +=
          4
              + Math.hypot(
                      n.route[(i + 1) % count][0] - n.route[i][0],
                      n.route[(i + 1) % count][1] - n.route[i][1])
                  / 18;
    double time = state.elapsed % duration;
    for (int i = 0; i < count; i++) {
      float[] a = n.route[i], b = n.route[(i + 1) % count];
      double travel = Math.hypot(b[0] - a[0], b[1] - a[1]) / 18;
      if (time < 4 + travel) {
        float t = (float) Math.max(0, (time - 4) / travel);
        out.x = a[0] + (b[0] - a[0]) * t;
        out.y = a[1] + (b[1] - a[1]) * t;
        out.direction =
            time < 4
                ? 0
                : Math.abs(b[0] - a[0]) > Math.abs(b[1] - a[1])
                    ? (b[0] < a[0] ? 1 : 2)
                    : (b[1] < a[1] ? 3 : 0);
        out.moving = time >= 4;
        return;
      }
      time -= 4 + travel;
    }
  }

  public Content.Npc nearestNpc() {
    Content.Npc found = null;
    double best = 28;
    Actor a = new Actor();
    for (var n : npcs()) {
      position(n, a);
      double dist = Math.hypot(state.x - a.x, state.y - a.y);
      if (dist < best) {
        best = dist;
        found = n;
      }
    }
    return found;
  }

  /** Nearby fixtures win over a passing NPC, consistently in the prompt and input handler. */
  public Content.Npc interactionNpc() {
    var prop = nearestProp();
    if (prop != null
        && Math.hypot(state.x - prop.x, state.y - prop.y) < 21
        && Set.of("door", "shop", "computer", "rest", "date").contains(prop.action)) return null;
    return nearestNpc();
  }

  public void enter(Content.Prop p) {
    if (!p.action.equals("door")) throw new IllegalArgumentException("Это не дверь");
    String old = state.map;
    float x = state.x, y = state.y;
    state.map = p.target;
    state.x = p.targetX;
    state.y = p.targetY;
    if (blocked(state.x, state.y)) {
      state.map = old;
      state.x = x;
      state.y = y;
      throw new IllegalStateException("Выход двери заблокирован");
    }
    if (!state.visited.contains(state.map)) state.visited.add(state.map);
    state.direction = p.target.equals("room") ? 3 : 0;
    moving = false;
  }

  public String npcDialogue(Content.Npc npc) {
    return switch (npc.id) {
      case "mira" -> state.hasLetter ? "miraFound" : state.deliveredLetter ? "miraAgain" : "mira";
      case "arkady" -> state.metArkady ? "arkadyAgain" : "arkady";
      default -> npc.dialogue;
    };
  }

  public String propDialogue(Content.Prop p) {
    return switch (p.action) {
      case "computer" -> "pc";
      case "plant" -> state.wateredPlant ? "plantAgain" : "plant";
      case "boxes" -> state.unpacked ? "boxesAgain" : "boxes";
      case "post" -> state.signedMailbox ? "postAgain" : "post";
      default -> p.action;
    };
  }

  public void effect(String effect) {
    switch (effect) {
      case "meetMira" -> state.metMira = true;
      case "meetArkady" -> state.metArkady = true;
      case "meetLev" -> state.metLev = true;
      case "inspectPc" -> state.inspectedPc = true;
      case "askLetter" -> {
        state.metMira = true;
        state.askedLetter = true;
      }
      case "takeLetter" -> {
        if (!state.deliveredLetter) state.hasLetter = true;
      }
      case "deliverLetter" -> {
        if (state.hasLetter) {
          state.hasLetter = false;
          state.deliveredLetter = true;
          state.metMira = true;
        }
      }
      case "waterPlant" -> state.wateredPlant = true;
      case "unpack" -> state.unpacked = true;
      case "signMailbox" -> state.signedMailbox = true;
      case "readBoard" -> state.readBoard = true;
      case "rest" -> state.elapsed += 30;
      case "water", "" -> {}
      default -> throw new IllegalArgumentException("Неизвестный эффект: " + effect);
    }
  }
}

package life.ctrl.save;

import com.badlogic.gdx.utils.Json;
import com.badlogic.gdx.utils.JsonWriter;
import java.io.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import life.ctrl.content.Content;
import life.ctrl.model.*;

/** Validated, atomic single-slot saves with a previous-good backup. */
public final class SaveStore {
  public static final int VERSION = 4;

  public static final class Envelope {
    public int version = VERSION;
    public GameState state;

    public Envelope() {}

    public Envelope(GameState s) {
      state = s;
    }
  }

  public record Loaded(GameState state, boolean fromBackup) {}

  private final Path directory;
  private final Content content;

  public SaveStore(Path directory, Content content) {
    this.directory = directory;
    this.content = content;
  }

  public static Path defaultDirectory() {
    String override = System.getProperty("ctrl.dataDir");
    if (override != null) return Path.of(override);
    String base =
        System.getProperty("os.name").startsWith("Windows")
            ? System.getenv("LOCALAPPDATA")
            : System.getenv("XDG_DATA_HOME");
    if (base == null || base.isBlank())
      base = Path.of(System.getProperty("user.home"), ".local", "share").toString();
    return Path.of(base, "CtrlLife");
  }

  private Json json() {
    Json j = new Json();
    j.setOutputType(JsonWriter.OutputType.json);
    j.setUsePrototypes(false);
    return j;
  }

  public boolean exists() {
    return Files.isRegularFile(directory.resolve("save.json"))
        || Files.isRegularFile(directory.resolve("save.backup.json"));
  }

  private GameState read(Path file) throws IOException {
    if (Files.size(file) > 65536) throw new IOException("Файл сохранения слишком большой");
    try {
      Envelope e = json().fromJson(Envelope.class, Files.readString(file, StandardCharsets.UTF_8));
      if (e == null || (e.version < 1 || e.version > VERSION) || e.state == null)
        throw new IllegalArgumentException("Версия сохранения не поддерживается");
      if (e.version == 1) {
        e.state.gameMinutes = 17 * 60 + 30 + e.state.elapsed / 2;
        var m = content.map(e.state.map);
        // The cutaway room and furniture changed. Old positions get a safe entrance.
        if (e.state.map.equals("room")) {
          e.state.x = m.spawnX;
          e.state.y = m.spawnY;
        }
        try {
          new World(content, e.state);
        } catch (IllegalArgumentException ex) {
          e.state.x = m.spawnX;
          e.state.y = m.spawnY;
        }
      }
      if (e.version < 3) {
        e.state.completedOrders = e.state.paidJobs.size();
        e.state.codingXp = e.state.completedOrders;
        for (var ledger : e.state.paidJobs) {
          String id = ledger.substring(ledger.indexOf(':') + 1);
          if (!e.state.shipped.contains(id)) e.state.shipped.add(id);
        }
      }
      if (e.version < 4) {
        var map = content.map(e.state.map);
        for (var prop : map.objects) {
          if (prop.id.startsWith("v4_")
              && prop.solid != null
              && World.intersects(e.state.x, e.state.y, prop.solid)) {
            e.state.x = map.spawnX;
            e.state.y = map.spawnY;
            break;
          }
        }
      }
      new World(content, e.state);
      return e.state;
    } catch (RuntimeException ex) {
      throw new IOException("Повреждённое или несовместимое сохранение", ex);
    }
  }

  public Loaded load() throws IOException {
    try {
      return new Loaded(read(directory.resolve("save.json")), false);
    } catch (IOException main) {
      try {
        return new Loaded(read(directory.resolve("save.backup.json")), true);
      } catch (IOException backup) {
        throw new IOException("Не удалось прочитать сохранение и резервную копию", main);
      }
    }
  }

  public void save(GameState state) throws IOException {
    new World(content, state);
    Files.createDirectories(directory);
    Path main = directory.resolve("save.json"),
        temp = directory.resolve("save.tmp.json"),
        backup = directory.resolve("save.backup.json");
    Files.writeString(temp, json().prettyPrint(new Envelope(state)), StandardCharsets.UTF_8);
    read(temp);
    if (Files.isRegularFile(main)) {
      boolean valid = true;
      try {
        read(main);
      } catch (IOException ignored) {
        valid = false;
      }
      if (valid) Files.copy(main, backup, StandardCopyOption.REPLACE_EXISTING);
    }
    replace(temp, main);
  }

  public static void replace(Path temp, Path file) throws IOException {
    try {
      Files.move(temp, file, StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING);
    } catch (AtomicMoveNotSupportedException ex) {
      Files.move(temp, file, StandardCopyOption.REPLACE_EXISTING);
    }
  }

  public Path directory() {
    return directory;
  }
}
