/**
 * High-resolution Scene2D implementation of the complete Workbench gameplay.
 * The rendered 480x270 PixelCanvas is deliberately not used by this application.
 * GameState, Coding and ByteScript are the single source of truth for its results.
 */
final class AstraCodeApp {
  private static final Color FG = Color.valueOf("dfe5ed");
  private static final Color MUTED = Color.valueOf("8c9aab");
  private static final Color BLUE = Color.valueOf("78b4ee");
  private static final Color GREEN = Color.valueOf("98c379");
  private static final Color GOLD = Color.valueOf("e5c07b");
  private static final Color ERROR = Color.valueOf("e18a94");

  private final CtrlLife game;
  private final LifeScreens life;
  private final Skin skin;
  private final Table root = new Table();
  private int displayedPage = -1, boardPage, portfolioPage;
  private String displayedKey = "", notice = "";
  private boolean dirty = true, rush;
  private double[] inputs = {0};
  private String labResult = "Измени аргументы и запусти собственный тест.";
  private int baseline = -1, operations = -1;

  AstraCodeApp(CtrlLife game, LifeScreens life, Skin skin) {
    this.game = game;
    this.life = life;
    this.skin = skin;
    root.top().left();
    root.setBackground(skin.getDrawable("pane-bg"));
  }

  Table view() { return root; }
  void invalidate() { dirty = true; }

  private GameState state() { return game.world().state; }
  private LifeContent.Job job() {
    return state().activeJob.isEmpty() ? null : life.data.job(state().activeJob);
  }
  private void persist() { game.refreshState(); }
  private void message(String text) { notice = text; dirty = true; }
  private void page(int n) { life.codePage = n; dirty = true; }
  private void leave() { persist(); game.switchMode(Mode.WORLD); }

  void tick() {
    if (job() == null && (life.codePage == 1 || life.codePage == 2 ||
        life.codePage == 3 || life.codePage == 4 || life.codePage == 5 ||
        life.codePage == 6 || life.codePage == 10)) life.codePage = 0;
    String report = life.report == null ? "" :
        life.report.compiled() + ":" + life.report.passed() + ":" + life.report.message();
    String key = life.codePage + "|" + state().activeJob + "|" + state().jobStage +
        "|" + state().typedCount + "|" + state().codeSource + "|" + report +
        "|" + state().money + "|" + state().codingXp + "|" + state().revisions.size() +
        "|" + state().shipped.size() + "|" + state().upgrades.size();
    if (dirty || life.codePage != displayedPage || !key.equals(displayedKey)) {
      displayedPage = life.codePage;
      displayedKey = key;
      dirty = false;
      rebuild();
    }
  }

  private Label label(String text, Color color, boolean small) {
    Label.LabelStyle style = new Label.LabelStyle(
        skin.getFont(small ? "small-font" : "ui-font"), color);
    Label result = new Label(text, style);
    result.setAlignment(Align.left);
    return result;
  }

  private Label wrap(String text, Color color) {
    Label label = label(text, color, true);
    label.setWrap(true);
    label.setAlignment(Align.topLeft);
    return label;
  }

  private TextButton button(String text, Runnable action, boolean enabled) {
    TextButton.TextButtonStyle style = new TextButton.TextButtonStyle();
    style.font = skin.getFont("small-font");
    style.fontColor = enabled ? FG : MUTED;
    style.overFontColor = Color.WHITE;
    style.up = skin.getDrawable("button-bg");
    style.over = skin.getDrawable("hover-bg");
    style.down = skin.getDrawable("hover-bg");
    TextButton btn = new TextButton(text, style);
    btn.setDisabled(!enabled);
    btn.getLabel().setAlignment(Align.left);
    btn.getLabel().setWrap(true);
    btn.padLeft(14);
    btn.padRight(11);
    if (enabled) btn.addListener(new ChangeListener() {
      @Override public void changed(ChangeEvent event, Actor actor) {
        action.run();
        dirty = true;
      }
    });
    return btn;
  }

  private void action(Table table, String caption, Runnable run) {
    action(table, caption, run, true);
  }
  private void action(Table table, String caption, Runnable run, boolean enabled) {
    table.add(button(caption, run, enabled)).height(43).growX().minWidth(0).padBottom(8);
    table.row();
  }

  private void note(Table table, String value) {
    Label label = wrap(value, MUTED);
    table.add(label).width(625).growX().minWidth(0).padBottom(16);
    table.row();
  }

  private void title(Table table, String category, String name, String subtitle) {
    Label eye = label(category.toUpperCase(Locale.ROOT), BLUE, true);
    table.add(eye).left().padBottom(12);
    table.row();
    table.add(label(name, FG, false)).left().padBottom(7);
    table.row();
    note(table, subtitle);
  }

  private Table block() {
    Table table = new Table();
    table.top().left().pad(17);
    table.setBackground(skin.getDrawable("input-bg"));
    return table;
  }

  private void addBlock(Table content, Table block) {
    content.add(block).growX().minWidth(0).padBottom(12);
    content.row();
  }

  private void header(Table bar) {
    bar.setBackground(skin.getDrawable("window-header"));
    bar.left().padLeft(17).padRight(18);
    Label logo = label("A  /  ASTRA CODE", BLUE, false);
    bar.add(logo).left().padRight(25);
    bar.add(label("ByteScript  •  project workspace", MUTED, true)).left();
    bar.add().expandX();
    bar.add(label(Career.rank(state()) + "   ·   " + state().codingXp + " XP", MUTED, true))
        .padRight(20);
    bar.add(label(state().money + " ₽", GOLD, true));
  }

  private void nav(Table side, String name, int page, boolean available) {
    boolean selected = life.codePage == page;
    TextButton b = button((selected ? "▎  " : "     ") + name, () -> navigate(page), available);
    // Desktop navigation uses a quiet flat surface, not the blue CTA capsules.
    // Only the active document has an accent, like Archcraft's compact menus.
    b.getStyle().up = skin.getDrawable(selected ? "term-tab" : "term-toolbar");
    b.getStyle().over = skin.getDrawable("term-tab");
    b.getStyle().down = skin.getDrawable("term-tab");
    b.getStyle().fontColor = selected ? BLUE : MUTED;
    b.getStyle().overFontColor = FG;
    side.add(b).height(36).growX().minWidth(0).padBottom(2);
    side.row();
  }

  private void navigate(int next) {
    if (next == 4) {
      if (job() == null) { page(0); return; }
      if (life.report == null || !life.report.passed()) {
        test();
        if (life.report == null || !life.report.passed()) {
          message("Сначала пройди тесты, затем открой предпросмотр.");
          page(2);
          return;
        }
      }
      life.previewInput = job().id.equals("discount") ? 500 :
          job().seed[job().seed.length - 1];
    }
    if (next == 5) prepareLab();
    page(next);
  }

  private void sidebar(Table side) {
    side.top().left().pad(12);
    side.setBackground(skin.getDrawable("window-bg"));
    Label work = label("WORKSPACE", BLUE, true);
    side.add(work).left().padLeft(11).padTop(8).padBottom(12);
    side.row();
    nav(side, "Контракты", 0, true);
    nav(side, "main.bs", 2, job() != null);
    nav(side, "Исправления", 3, job() != null);
    nav(side, "Предпросмотр", 4, job() != null);
    nav(side, "Лаборатория", 5, job() != null);
    nav(side, "Git / версии", 6, job() != null);
    side.add(label("PROGRESSION", BLUE, true)).padLeft(11).padTop(16).padBottom(10).left();
    side.row();
    nav(side, "Карьера", 7, true);
    nav(side, "Портфолио", 11, true);
    nav(side, "Оборудование", 8, true);
    nav(side, "Свой бот", 9, true);
    side.add().expandY();
    side.row();
    action(side, "Выйти из ПК", this::leave);
  }

  private void rebuild() {
    root.clearChildren();
    Table bar = new Table();
    header(bar);
    root.add(bar).height(53).growX();
    root.row();

    Table body = new Table();
    body.top().left();
    Table side = new Table();
    sidebar(side);
    body.add(side).width(208).growY().top();
    Table content = new Table();
    content.top().left().pad(22);
    content.defaults().left();
    renderPage(content);
    if (!notice.isEmpty()) {
      Table info = block();
      Label message = wrap(notice, GOLD);
      info.add(message).width(600).growX().minWidth(0);
      addBlock(content, info);
    }
    ScrollPane pane = new ScrollPane(content, new ScrollPane.ScrollPaneStyle());
    pane.setFadeScrollBars(true);
    pane.setOverscroll(false, false);
    pane.setScrollingDisabled(true, false);
    body.add(pane).grow().minWidth(0);
    root.add(body).grow().minHeight(0);
  }

  private void renderPage(Table content) {
    LifeContent.Job j = job();
    switch (life.codePage) {
      case 0 -> board(content);
      case 1 -> brief(content, j);
      case 2 -> editor(content, j);
      case 3 -> patches(content, j);
      case 4 -> preview(content, j);
      case 5 -> lab(content, j);
      case 6 -> history(content);
      case 7 -> career(content);
      case 8 -> gear(content);
      case 9 -> project(content);
      case 10 -> abandon(content);
      case 11 -> portfolio(content);
      default -> board(content);
    }
  }

  private void choose(LifeContent.Job j) {
    if (!state().activeJob.isEmpty() && state().jobStage >= 2 && state().jobStage < 5) {
      message("Текущий исходник сохранён. Закончи заказ или отложи его через Git.");
      page(2);
      return;
    }
    String reason = Career.blocked(state(), j);
    if (!reason.isEmpty()) { message(reason); return; }
    state().activeJob = j.id;
    rush = false;
    life.report = null;
    notice = "";
    page(1);
    persist();
  }

  private void board(Table t) {
    title(t, "TASKMAIL / JOB BOARD", "Биржа заказов",
        "День " + state().day() + "  ·  оплачено " + Career.paidCountToday(state()) +
        " из 3 клиентских заказов сегодня. Выбирай задачу, читай ТЗ, пиши код и проверяй тесты.");
    var offers = Career.offers(state(), life.data);
    int pages = Math.max(1, (offers.size() + 3) / 4);
    boardPage = Math.floorMod(boardPage, pages);
    for (int i = boardPage * 4; i < Math.min(offers.size(), boardPage * 4 + 4); i++) {
      LifeContent.Job j = offers.get(i);
      String reason = Career.blocked(state(), j);
      Table card = block();
      card.add(label(j.client + "   /   " + j.title, FG, false)).growX().left();
      card.row();
      card.add(label(j.pay + " ₽   •   " + Career.duration(state(), j) + " мин   •   " +
          (reason.isEmpty() ? "Доступно" : reason), reason.isEmpty() ? GREEN : MUTED, true))
          .padTop(7).padBottom(10).left();
      card.row();
      action(card, reason.isEmpty() ? "Открыть заказ" : "Недоступно",
          () -> choose(j), reason.isEmpty());
      addBlock(t, card);
    }
    Table switcher = new Table();
    switcher.add(button("←", () -> { boardPage = Math.floorMod(boardPage - 1, pages); }, pages > 1))
        .width(70).height(42);
    switcher.add(label("   Страница " + (boardPage + 1) + " / " + pages + "   ", MUTED, true));
    switcher.add(button("→", () -> { boardPage = (boardPage + 1) % pages; }, pages > 1))
        .width(70).height(42);
    t.add(switcher).left();
    t.row();
  }

  private void brief(Table t, LifeContent.Job j) {
    title(t, "CONTRACT / " + j.id, j.title, j.client + "   •   " + j.pay + " ₽");
    Table card = block();
    card.add(wrap(j.brief, FG)).width(610).growX().minWidth(0).padBottom(16);
    card.row();
    card.add(label("Время: " + Career.duration(state(), j) + " мин   /   запуск: " +
        j.cost + " ₽", GOLD, true)).left();
    card.row();
    if (!j.once && state().completedOrders > 0) {
      action(card, rush ? "● Срочный заказ (+25%)" : "○ Срочный заказ (+25%)",
          () -> rush = !rush);
    }
    if (j.kind.equals("feature")) {
      card.add(label("ВЫБЕРИ РЕАЛИЗАЦИЮ", BLUE, true)).left().padTop(15);
      card.row();
      for (int i = 0; i < j.algorithms.length; i++) {
        int choice = i;
        action(card, j.algorithms[i], () -> start(j, choice));
      }
    } else {
      action(card, j.kind.equals("profile") ? "Открыть профилировщик и код" :
          "Открыть исходник", () -> start(j, 0));
      action(card, "Подсказка клиента", () -> message(j.hint));
    }
    addBlock(t, card);
    action(t, "← Вернуться на биржу", () -> page(0));
  }

  private void start(LifeContent.Job j, int choice) {
    String result = Coding.accept(state(), j, choice, rush, true);
    if (state().jobStage < 2 || !state().activeJob.equals(j.id)) {
      message(result);
      return;
    }
    life.report = null;
    notice = "";
    page(2);
    persist();
  }

  private void editor(Table t, LifeContent.Job j) {
    title(t, "EDITOR / SRC", j.title,
        "main.bs   ·   ByteScript   /   Нажимай буквенные клавиши: игровой код печатается автоматически.");
    Table code = block();
    code.add(label("◇  main.bs", BLUE, true)).left().padBottom(13);
    code.row();
    String source = state().codeSource;
    int typed = Math.max(0, Math.min(state().typedCount, source.length()));
    String[] lines = source.substring(0, typed).split("\n", -1);
    int first = Math.max(0, lines.length - 18);
    for (int i = first; i < lines.length; i++) {
      Table row = new Table();
      row.left();
      row.add(label(String.format("%02d   ", i + 1), MUTED, true)).width(55).left();
      String codeLine = lines[i];
      String trim = codeLine.stripLeading();
      Color syntax = trim.startsWith("fn ") ? BLUE :
          trim.startsWith("return ") ? GOLD :
          trim.startsWith("//") ? MUTED : GREEN;
      row.add(label(codeLine + (i == lines.length - 1 ? " ▏" : ""), syntax, true))
          .left().growX();
      code.add(row).left().growX().minWidth(0).padBottom(4);
      code.row();
    }
    code.add(label("Символов: " + typed + " / " + source.length() +
        "  •  " + (source.isEmpty() ? 100 :
            Math.round(typed * 100.0 / source.length())) + "%", GOLD, true))
        .padTop(17).left();
    addBlock(t, code);

    Table console = block();
    console.add(label("ПРОБЛЕМЫ   /   ВЫВОД   /   ТЕСТЫ", BLUE, true)).left().padBottom(11);
    console.row();
    if (life.report == null) {
      console.add(wrap(typed < source.length() ?
          "РЕЖИМ ВВОДА • нажимай клавиши, чтобы допечатать исходник." :
          "Код готов. Запусти компиляцию и тесты.", MUTED))
          .width(610).growX().minWidth(0);
    } else {
      console.add(wrap((life.report.passed() ? "PASS • " : "FAIL • ") +
          life.report.message(), life.report.passed() ? GREEN : ERROR))
          .width(610).growX().minWidth(0);
      console.row();
      for (String result : life.report.cases()) {
        console.add(wrap(result, result.endsWith("OK") ? GREEN : ERROR))
            .width(610).growX().minWidth(0).padTop(5);
        console.row();
      }
    }
    addBlock(t, console);
    Table commands = new Table();
    commands.defaults().padRight(7);
    commands.add(button("▶ Компиляция / тесты", this::test, true)).height(43).growX();
    commands.add(button("Исправления", () -> page(3), true)).height(43).growX();
    commands.add(button("Предпросмотр", () -> navigate(4),
        life.report != null && life.report.passed())).height(43).growX();
    t.add(commands).growX().minWidth(0);
    t.row();
    Table extras = new Table();
    extras.defaults().padRight(7);
    extras.add(button("Лаборатория", () -> navigate(5), true)).height(43).growX();
    extras.add(button("Git / версии", () -> page(6), true)).height(43).growX();
    t.add(extras).growX().minWidth(0).padTop(7);
    t.row();
  }

  private void test() {
    if (job() == null) return;
    life.report = Coding.test(state(), job());
    message((life.report.passed() ? "✓ " : "✕ ") + life.report.message());
    persist();
  }

  private void patches(Table t, LifeContent.Job j) {
    title(t, "DEBUGGER / CODE REVIEW", "Исправление ошибок",
        "Выбери изменение и заново напечатай код. Ошибки компиляции проверяются настоящим ByteScript.");
    note(t, state().codeSource.contains("prce") ?
        "Имя prce не объявлено. В аргументах есть price и count — что должно участвовать в произведении?"
        : j.hint);
    for (int i = 0; i < j.patches.length; i++) {
      int choice = i;
      action(t, j.patches[i], () -> {
        Coding.repair(state(), j, choice, true);
        life.report = null;
        notice = "Патч применён. Допечатай код и повтори тесты.";
        page(2);
        persist();
      });
    }
    action(t, "← Вернуться к исходнику", () -> page(2));
  }

  private void preview(Table t, LifeContent.Job j) {
    title(t, "RUN / LIVE PREVIEW", "Предпросмотр",
        "Результат выполняется игровым интерпретатором ByteScript. Перед отправкой обязательно запусти его.");
    if (life.report == null || !life.report.passed()) {
      note(t, "Чтобы открыть предпросмотр, сначала запусти тесты в редакторе.");
      action(t, "К тестам", () -> page(2));
      return;
    }
    double value;
    try {
      value = Coding.preview(state(), j, life.previewInput);
    } catch (RuntimeException e) {
      note(t, "Ошибка запуска: " + e.getMessage());
      action(t, "К исходнику", () -> page(2));
      return;
    }
    Table result = block();
    result.add(label("INPUT   →   OUTPUT", BLUE, true)).left().padBottom(16);
    result.row();
    result.add(label(Coding.format(life.previewInput) + "   →   " +
        Coding.format(value), GREEN, false)).left().padBottom(16);
    result.row();
    Table controls = new Table();
    controls.add(button("−10", () -> life.previewInput = Math.max(-10000, life.previewInput - 10), true))
        .width(76).height(40).padRight(5);
    controls.add(button("−1", () -> life.previewInput = Math.max(-10000, life.previewInput - 1), true))
        .width(76).height(40).padRight(5);
    controls.add(button("+1", () -> life.previewInput = Math.min(10000, life.previewInput + 1), true))
        .width(76).height(40).padRight(5);
    controls.add(button("+10", () -> life.previewInput = Math.min(10000, life.previewInput + 10), true))
        .width(76).height(40);
    result.add(controls).left();
    addBlock(t, result);
    note(t, "Время работы: " + Career.duration(state(), j) + " мин. " +
        "Премия: " + Career.bonus(state(), j) + " ₽.");
    action(t, j.once ? "Запустить этап • −" + j.cost + " ₽" :
        "Сдать клиенту • " + (j.pay + Career.bonus(state(), j)) + " ₽",
        () -> deliver(j), state().jobStage != 5);
    action(t, "← Вернуться к исходнику", () -> page(2));
  }

  private void deliver(LifeContent.Job j) {
    String result = Coding.deliver(state(), j);
    if (state().jobStage == 5) {
      persist();
      game.successCue();
      life.scene(j.once ? "Этап запущен" : "Работа принята", result, Mode.WORLD);
    } else message(result);
  }

  private int operations(String source, LifeContent.Job j) {
    try {
      var program = ByteScript.compile(source);
      int max = 0;
      for (var item : j.tests) max = Math.max(max, program.operations(item.args));
      return max;
    } catch (RuntimeException e) {
      return -1;
    }
  }

  private void measure(LifeContent.Job j) {
    baseline = operations(state().revisions.isEmpty() ? state().codeSource :
        state().revisions.get(0).source, j);
    operations = operations(state().codeSource, j);
  }

  private void prepareLab() {
    if (job() == null) return;
    inputs = job().seed.clone();
    labResult = "Проверь ноль, границы и отрицательные числа, затем запусти.";
    measure(job());
  }

  private void lab(Table t, LifeContent.Job j) {
    title(t, "TOOLS / PROFILER", "Лаборатория",
        "Свои тестовые случаи приносят бонус за качество. Ограничение — 10 000 по модулю.");
    var params = ByteScript.compile(j.source).parameters();
    for (int i = 0; i < inputs.length; i++) {
      int index = i;
      Table row = block();
      row.add(label(params.get(i) + " = " + Coding.format(inputs[i]), FG, false))
          .width(180).left().padRight(12);
      for (int step : new int[] {-10, -1, 1, 10}) {
        row.add(button((step > 0 ? "+" : "") + step, () -> {
          inputs[index] = Math.max(-10000, Math.min(10000, inputs[index] + step));
        }, true)).width(64).height(39).padRight(5);
      }
      addBlock(t, row);
    }
    note(t, "Операций: было " + (baseline < 0 ? "—" : baseline) +
        " → сейчас " + (operations < 0 ? "ошибка" : operations) +
        (j.operationBudget > 0 ? "  /  лимит " + j.operationBudget : "") +
        "     •     собственные тесты " + Math.min(2, state().customCases.size()) + " / 2");
    Table box = block();
    box.add(wrap(labResult, FG)).width(600).growX().minWidth(0);
    addBlock(t, box);
    action(t, "▶ Запустить свой тест", () -> {
      labResult = Coding.probe(state(), j, inputs.clone());
      measure(j);
      persist();
    });
    action(t, "Подсказка клиента", () -> message(j.hint));

  private void map() {

        w = m.width * ratio,
        h = m.height * ratio,
        ox = 25 + (300 - w) / 2,
        oy = 65 + (151 - h) / 2;
    c.scaledArt(id, ox, oy, w, h);
    var boundary = life.ctrl.render.SceneLayout.bounds(id);
    c.clip(
        ox + boundary.x() * ratio,
        oy + boundary.y() * ratio,
        boundary.width() * ratio,
        boundary.height() * ratio);
    Arrays.stream(m.objects)
        .filter(p -> p.art != null && game.world().visible(p))
        .sorted(
            Comparator.comparingDouble(
                p -> p.solid != null ? p.solid[1] + p.solid[3] : p.artY + c.artHeight(p.art)))
        .forEach(
            p -> {
              String art = p.art;
              if (p.id.equals("boxes") && s().unpacked) art = "boxes_open";
              if (p.id.equals("post") && s().signedMailbox) art = "mailbox_signed";
              if (p.art.equals("desk") && id.equals("room") && s().upgrades.contains("ssd"))
                art = "desk_ssd";
              c.scaledArt(
                  art,
                  ox + p.artX * ratio,
                  oy + p.artY * ratio,
                  c.artWidth(art) * ratio,
                  c.artHeight(art) * ratio);
            });
    c.unclip();
    for (var p : m.objects)
      if (p.action.equals("door")) {
        float x = ox + p.x * ratio, y = oy + p.y * ratio;
        c.rect(x - 2, y - 2, 5, 5, GOLD);

  private NinePatchDrawable patch(String id, int r, int red, int green, int blue,
                                  int alpha, int unused) {
    // NinePatch texture created at 3x target radius to keep silhouettes smooth.
    int size = 96, border = 32, radius = Math.min(38, r * 2);
    Pixmap pix = new Pixmap(size, size, Pixmap.Format.RGBA8888);
    for (int y = 0; y < size; y++) {
      float ay = Math.abs(y + .5f - size / 2f) - (size / 2f - radius);
      for (int x = 0; x < size; x++) {
        float ax = Math.abs(x + .5f - size / 2f) - (size / 2f - radius);
        float distance = (float)Math.hypot(Math.max(ax, 0), Math.max(ay, 0))
            + Math.min(0, Math.max(ax, ay)) - radius;
        float a = Math.max(0, Math.min(1, .5f - distance)) * alpha / 255f;
        pix.drawPixel(x, y, (red << 24) | (green << 16) | (blue << 8) | Math.round(a * 255));
      }
    }
    Texture t = new Texture(pix);
    t.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    pix.dispose();
    skin.add(id + "-texture", t);
    NinePatch patch = new NinePatch(t, border, border, border, border);
    return new NinePatchDrawable(patch);
  }

    window.addCaptureListener(new InputListener() {

                                    int pointer, int button) {
        resizing = false;
      }

    maximized.put(app, false);
    return window;

  void open(App app) {
    VisWindow w = windows.get(app);
    if (w == null) w = create(app);
    if (w.getStage() == null) stage.addActor(w);
    w.setVisible(true);
    if (app == App.CODE && codeApp != null) codeApp.invalidate();
    if (app != App.TERMINAL) stage.setKeyboardFocus(null);
    w.setTouchable(com.badlogic.gdx.scenes.scene2d.Touchable.enabled);
    w.toFront();
    w.clearActions();
    w.getColor().a = 0f;
    w.addAction(Actions.fadeIn(.16f));
    if (app == App.TERMINAL && terminalInput != null) {
      terminalOutput.setText(shell.terminalHistory());
      stage.setKeyboardFocus(terminalInput);
    }

  void close(App app) {
    VisWindow w = windows.get(app);
    if (w == null) return;
    stage.unfocus(w);
    w.clearActions();
    w.setTouchable(com.badlogic.gdx.scenes.scene2d.Touchable.disabled);
    w.addAction(Actions.sequence(Actions.fadeOut(.12f), Actions.visible(false)));
  }

  void minimize(App app) {
    VisWindow w = windows.get(app);
    if (w == null) return;
    w.setVisible(false);
    stage.unfocus(w);
  }

  void closeFront() {
    Actor top = front();
    for (var entry : windows.entrySet())
      if (entry.getValue() == top) { close(entry.getKey()); return; }
  }

  Actor front() {
    Actor result = null;
    int index = -1;
    for (var e : windows.entrySet()) {
      VisWindow w = e.getValue();
      if (w.getStage() == null || !w.isVisible()) continue;
      int i = stage.getRoot().getChildren().indexOf(w, true);
      if (i > index) { index = i; result = w; }
    }
    return result;
  }

  boolean hasOpen() { return front() != null; }

  boolean isFrontCode() { return front() != null && front() == windows.get(App.CODE); }

  void tick() { if (codeApp != null && isVisible(App.CODE)) codeApp.tick(); }

  void cycleFocus() {
    Actor current = front();
    VisWindow candidate = null;
    int lowest = Integer.MAX_VALUE;
    for (VisWindow w : windows.values()) {
      if (w.getStage() != null && w.isVisible() && w != current) {
        int index = stage.getRoot().getChildren().indexOf(w, true);
        if (index < lowest) { lowest = index; candidate = w; }
      }
    }
    if (candidate != null) candidate.toFront();
  }

  boolean isVisible(App app) {
    VisWindow w = windows.get(app);
    return w != null && w.isVisible();
  }

  void toggleMaximize(App app) {
    VisWindow w = windows.get(app);
    if (w == null) return;
    boolean isMax = maximized.getOrDefault(app, false);
    if (!isMax) {
      restored.put(app, new float[] {w.getX(), w.getY(), w.getWidth(), w.getHeight()});
      float h = stage.getHeight(), width = stage.getWidth();
      w.setBounds(24, 102, width - 48, h - 174);
    } else {
      float[] p = restored.get(app);
      if (p != null) w.setBounds(p[0], p[1], p[2], p[3]);
    }
    maximized.put(app, !isMax);
    w.invalidateHierarchy();
  }

  void resize(int width, int height) {
    for (VisWindow w : windows.values()) {
      w.setPosition(
          Math.max(2, Math.min(w.getX(), width - w.getWidth() - 8)),
          Math.max(95, Math.min(w.getY(), height - w.getHeight() - 52)));
    }
  }

  void reset() {
    for (App app : windows.keySet()) {
      VisWindow w = windows.get(app);
      w.clearActions();
      w.setVisible(false);
      w.setColor(1, 1, 1, 1);
    }
  }

package life.ctrl.ui;

import com.badlogic.gdx.graphics.Pixmap;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import java.util.Random;

/**
 * A self-contained, deterministic Pinux desktop wallpaper. Native libGDX texture,
 * generated once on the rendering thread; no downloaded art or web view.
 *
 * 960x540 source is deliberately 2x the logical 480x270 game canvas so silhouettes
 * and light gradients survive high-DPI scaling while keeping the image inexpensive.
 */
public final class PinuxWallpaper {
  private static final int W = 960, H = 540;
  private Texture city;
  private final Random random = new Random(0xA57A21);

  public Texture texture() {
    if (city == null) create();
    return city;
  }

  public void draw(SpriteBatch batch, boolean duskTheme) {
    if (city == null) create();
    batch.setColor(1f, duskTheme ? .89f : 1f, duskTheme ? .79f : 1f, 1f);
    batch.draw(city, 0, 0, 480, 270);
    batch.setColor(1f, 1f, 1f, 1f);
  }

  private void create() {
    Pixmap p = new Pixmap(W, H, Pixmap.Format.RGBA8888);
    paintSky(p);
    paintClouds(p);
    paintSun(p);
    // Quiet skyline kept to the lower half: room for icons and application windows.
    // Three intentionally sparse parallax layers instead of a wall of identical towers.
    paintCity(p, 410, 33, 31, 16, 0x67436e, 0x533862, .17f);
    paintCity(p, 469, 47, 63, 22, 0x332f59, 0x252742, .16f);
    paintCity(p, 513, 68, 49, 16, 0x1b2640, 0x171e35, .13f);
    paintForeground(p);
    city = new Texture(p);
    city.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    p.dispose();
  }

  private void paintSky(Pixmap p) {
    final float[][] stops = {
      {0f, 12, 19, 49},
      {.26f, 54, 37, 89},
      {.49f, 117, 66, 136},
      {.63f, 215, 105, 146},
      {.76f, 173, 86, 137},
      {1f, 35, 39, 79}
    };
    for (int y = 0; y < H; y++) {
      float t = y / (float)(H - 1);
      int band = 0;
      while (band < stops.length - 2 && t > stops[band + 1][0]) band++;
      float[] a = stops[band], b = stops[band + 1];
      float v = Math.max(0, Math.min(1, (t - a[0]) / (b[0] - a[0])));
      v = v * v * (3 - 2 * v);
      for (int x = 0; x < W; x++) {
        float dist = (float) Math.sqrt(
            Math.pow((x - 730) / 320f, 2) + Math.pow((y - 288) / 155f, 2));
        float glow = Math.max(0, 1f - dist);
        glow = glow * glow * .24f;
        float grain = (((x * 17 + y * 29) % 19) - 9) * .095f;
        int r = (int) (mix(a[1], b[1], v) * (1 - glow) + 250 * glow + grain);
        int g = (int) (mix(a[2], b[2], v) * (1 - glow) + 153 * glow + grain);
        int blue = (int) (mix(a[3], b[3], v) * (1 - glow) + 150 * glow + grain);
        p.drawPixel(x, y, rgba(r, g, blue, 255));
      }
    }
  }

  private void paintClouds(Pixmap p) {
    // Painterly wisps (broad, translucent and slightly curved), never random scanlines.
    for (int i = 0; i < 8; i++) {
      int centerX = 120 + random.nextInt(800);
      int centerY = 48 + random.nextInt(210);
      int span = 65 + random.nextInt(145);
      int tint = centerY > 145 ? 0xdf83ab : 0x66509b;
      int rr = (tint >> 16) & 255, gg = (tint >> 8) & 255, bb = tint & 255;
      for (int dx = -span; dx <= span; dx++) {
        float along = Math.abs(dx) / (float)span;
        float edge = Math.max(0f, 1f - along * along);
        int y0 = centerY + (int)(3 * Math.sin(dx / 42.0 + i));
        int x = centerX + dx;
        if (x < 0 || x >= W) continue;
        for (int dy = -3; dy <= 3; dy++) {
          int y = y0 + dy;
          if (y < 0 || y >= H) continue;
          float falloff = (1f - Math.abs(dy) / 4f) * edge;
          float alpha = falloff * .12f;
          p.drawPixel(x, y, blend(p.getPixel(x, y), rgba(rr, gg, bb, 255), alpha));
        }
      }
    }
  }

  private void paintSun(Pixmap p) {
    int cx = 745, cy = 311, radius = 46;
    for (int y = cy - radius; y <= cy + radius; y++) {
      for (int x = cx - radius; x <= cx + radius; x++) {
        if (x < 0 || y < 0 || x >= W || y >= H) continue;
        float dist = (float)Math.hypot(x - cx, y - cy);
        if (dist >= radius) continue;
        float t = (y - (cy - radius)) / (radius * 2f);
        int r = 255, g = (int)mix(200, 132, t), b = (int)mix(182, 171, t);
        // Continuous glowing disk: deliberately avoid dark scanlines that read as bugs.
        float rim = Math.min(1f, (radius - dist) / 3.5f);
        p.drawPixel(x, y, blend(p.getPixel(x, y), rgba(r, g, b, 255), rim * .86f));
      }
    }
  }

  private void paintCity(Pixmap p, int horizon, int minWidth, int maxHeight, int variance,
                         int topColor, int bottomColor, float windows) {
    int x = -24;
    while (x < W) {
      int width = minWidth + 9 + random.nextInt(minWidth + 25);
      int roof = horizon - (10 + random.nextInt(maxHeight) + random.nextInt(variance));
      // Leave a glowing pocket around the setting sun.
      if (x < 810 && x + width > 665 && horizon < 500) roof += 19;
      roof = Math.max(230, Math.min(horizon - 12, roof));
      int r1 = (topColor >>> 16) & 255, g1 = (topColor >>> 8) & 255, b1 = topColor & 255;
      int r2 = (bottomColor >>> 16) & 255, g2 = (bottomColor >>> 8) & 255, b2 = bottomColor & 255;
      int colorRoof = rgba(r1, g1, b1, 255);
      // Small architectural variations: setback roofs, chimneys, narrow towers.
      int type = random.nextInt(5);
      p.setColor(colorRoof);
      if (type == 0) {
        p.fillRectangle(x + 7, roof - 7, Math.max(8, width - 14), 7);
      } else if (type == 1) {
        p.fillRectangle(x + width / 2, roof - 12, 2, 12);
        p.fillRectangle(x + width / 2 - 2, roof - 2, 6, 2);
      } else if (type == 2) {
        p.fillRectangle(x + 5, roof - 4, Math.max(9, width / 3), 4);
      }
      for (int yy = roof; yy <= H - 30; yy++) {
        float t = Math.max(0f, Math.min(1f, (yy - roof) / (float)(H - roof)));
        // Slightly lighter left façade gives buildings actual depth.
        int col = rgba((int)mix(r1, r2, t), (int)mix(g1, g2, t),
            (int)mix(b1, b2, t), 255);
        p.setColor(col);
        p.fillRectangle(x, yy, width, 1);
      }
      p.setColor(rgba(Math.min(255, r1 + 11), Math.min(255, g1 + 7),
          Math.min(255, b1 + 12), 255));
      p.fillRectangle(x, roof + 2, 2, H - roof - 35);
      int windowStep = minWidth >= 65 ? 14 : minWidth >= 47 ? 12 : 10;
      for (int wy = roof + 13; wy < H - 34; wy += 14 + random.nextInt(4)) {
        for (int wx = x + 9; wx < x + width - 6; wx += windowStep) {
          if (wx < 1 || wx + 2 >= W || random.nextFloat() > windows) continue;
          int col = random.nextBoolean()
              ? rgba(242, 165, 152, 255) : rgba(178, 135, 192, 255);
          p.setColor(col);
          p.fillRectangle(wx, wy, 2, 4);
        }
      }
      x += width + 3 + random.nextInt(10);
    }
  }

  private void paintForeground(Pixmap p) {
    // Dark architectural framing and layered near-field foliage. Visible only in
    // the peripheral wallpaper so it does not compete with application windows.
    p.setColor(rgba(13, 20, 39, 255));
    p.fillRectangle(0, 480, W, 60);
    p.setColor(rgba(79, 62, 102, 255));
    p.fillRectangle(0, 480, W, 3);
    for (int i = 0; i < 18; i++) {
      int x = i * 58;
      int y = 482 + random.nextInt(27);
      p.setColor(rgba(28 + i % 3 * 2, 28, 51, 255));
      p.fillRectangle(x, y, 1 + random.nextInt(3), H - y);
    }
    p.setColor(rgba(17, 21, 42, 255));
    p.fillRectangle(0, 0, 17, H);
    p.fillRectangle(W - 18, 0, 18, H);

    // Keep the sky uncluttered: no cartoon tree blobs in the upper corners.
    p.setColor(rgba(15, 20, 39, 255));
    p.fillRectangle(0, 475, W, 9);
  }

  private void drawBranch(Pixmap p, int x, int y, int length, int direction, int depth) {
    int sign = direction == 0 ? 1 : -1;
    p.setColor(rgba(19, 20, 40, 255));
    for (int i = 0; i < length; i++) {
      int xx = x + sign * i / 3;
      int yy = y - i;
      if (xx < 18 || xx >= W - 18 || yy < 15) break;
      p.fillRectangle(xx, yy, Math.max(1, 6 - i / 33), 3);
    }
    for (int i = 0; i < 11 + depth; i++) {
      int bx = x + sign * (12 + random.nextInt(85));
      int by = y - 20 - random.nextInt(length - 30);
      p.setColor(rgba(22 + i % 3 * 6, 24, 49 + i % 3 * 6, 255));
      p.fillCircle(bx, by, 5 + random.nextInt(12));
    }
  }

  private static float mix(float a, float b, float t) { return a + (b - a) * t; }

  private static int rgba(int r, int g, int b, int a) {
    return (clamp(r) << 24) | (clamp(g) << 16) | (clamp(b) << 8) | clamp(a);
  }

  private static int clamp(int v) { return Math.max(0, Math.min(255, v)); }

  private static int blend(int base, int overlay, float alpha) {
    float o = Math.max(0, Math.min(1, alpha));
    return rgba(
        (int)mix((base >>> 24) & 255, (overlay >>> 24) & 255, o),
        (int)mix((base >>> 16) & 255, (overlay >>> 16) & 255, o),
        (int)mix((base >>> 8) & 255, (overlay >>> 8) & 255, o), 255);
  }

  public void dispose() {
    if (city != null) {
      city.dispose();
      city = null;
    }
  }

package life.ctrl.ui;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.EnumMap;
import java.util.List;

/**
 * Windowing rules for the fictional Pinux Astra desktop. Coordinates use the
 * 480x270 logical canvas, not the host OS pixel coordinates.
 */
public final class PinuxWindowManager {
  public enum App { CODE, TERMINAL, FILES, MAIL, SETTINGS }
  public static final int TOP = 17, BOTTOM = 250;

  public static final class Window {
    public final App app;
    public final String title;
    public int x, y, width, height;
    public int restoreX, restoreY, restoreWidth, restoreHeight;
    public boolean visible, minimized, maximized;
    public float animation;
    private Window(App app, String title, int x, int y, int width, int height) {
      this.app = app;
      this.title = title;
      this.x = x; this.y = y; this.width = width; this.height = height;
      restoreX = x; restoreY = y; restoreWidth = width; restoreHeight = height;
    }
    public int right() { return x + width; }
    public int bottom() { return y + height; }
    public int contentY() { return y + 20; }
    public int contentHeight() { return height - 23; }
  }

  @FunctionalInterface
  public interface ContentPainter { void paint(Window window, boolean active); }

  private final EnumMap<App, Window> windows = new EnumMap<>(App.class);
  private final ArrayList<Window> zOrder = new ArrayList<>();
  private Window dragging, resizing;
  private int grabX, grabY, dragStartX, dragStartY;
  private int resizeWidth, resizeHeight;
  private float time;
  private long lastClickAt;
  private App lastClickApp;
  private boolean moved;
  private int hoverX = -1, hoverY = -1;
  private static final String EDGE = "4e7087", TEXT = "dceaf4", ACCENT = "8dcce0";

  public PinuxWindowManager() {
    add(App.CODE, "Astra Code  |  main.bs", 28, 27, 420, 202);
    add(App.TERMINAL, "Terminal  |  dev@astra", 99, 40, 353, 181);
    add(App.FILES, "Files  |  /home/dev", 49, 34, 370, 193);
    add(App.MAIL, "TaskMail  |  inbox", 74, 35, 352, 194);
    add(App.SETTINGS, "Settings  |  Appearance", 115, 48, 296, 176);
  }

  private void add(App app, String title, int x, int y, int w, int h) {
    windows.put(app, new Window(app, title, x, y, w, h));
  }

  public Window get(App app) { return windows.get(app); }
  public int openCount() { return (int) zOrder.stream().filter(w -> !w.minimized && w.visible).count(); }

  public Window front() {
    for (int i = zOrder.size() - 1; i >= 0; i--) {
      Window w = zOrder.get(i);
      if (w.visible && !w.minimized) return w;
    }
    return null;
  }

  public boolean isFront(App app) { Window w = front(); return w != null && w.app == app; }
  public boolean isMaximizedCode() {
    Window w = front();
    return w != null && w.app == App.CODE && w.maximized;
  }

  public void open(App app) {
    Window w = get(app);
    if (!w.visible) {
      w.animation = 0;
      w.visible = true;
    }
    w.minimized = false;
    focus(app);
  }

  public void focus(App app) {
    Window w = get(app);
    if (w == null || !w.visible) return;
    w.minimized = false;
    zOrder.remove(w);
    zOrder.add(w);
  }

  public void minimize(App app) {
    Window w = get(app);
    if (w == null) return;
    w.minimized = true;
    if (dragging == w) dragging = null;
    if (resizing == w) resizing = null;
  }

  public void close(App app) {
    Window w = get(app);
    if (w == null) return;
    w.visible = false;
    w.minimized = false;
    // Let the window shrink away instead of disappearing in a single frame.
    if (dragging == w) dragging = null;
    if (resizing == w) resizing = null;
  }

  public void closeFront() { Window w = front(); if (w != null) close(w.app); }

  public void toggleMaximize(App app) {
    Window w = get(app);
    if (w == null) return;
    if (!w.maximized) {
      w.restoreX = w.x; w.restoreY = w.y;
      w.restoreWidth = w.width; w.restoreHeight = w.height;
      w.x = 10; w.y = TOP + 1; w.width = 460; w.height = BOTTOM - TOP - 1;
      w.maximized = true;
    } else {
      w.x = w.restoreX; w.y = w.restoreY;
      w.width = w.restoreWidth; w.height = w.restoreHeight;
      w.maximized = false;
      clamp(w);
    }
  }

  /** Handles window caption, system buttons, focus and dragging. */
  public boolean pointerDown(int x, int y) {
    if (x < 0 || x >= PixelCanvas.W || y < TOP || y >= BOTTOM) return false;
    for (int i = zOrder.size() - 1; i >= 0; i--) {
      Window w = zOrder.get(i);
      if (!w.visible || w.minimized || !inside(w, x, y)) continue;
      boolean wasFront = front() == w;
      focus(w.app);
      if (!w.maximized && x >= w.right() - 10 && y >= w.bottom() - 10) {
        resizing = w;
        dragStartX = x; dragStartY = y;
        resizeWidth = w.width; resizeHeight = w.height;
        return true;
      }
      if (y < w.y + 19) {
        // Three caption buttons: minimize, maximize and close.
        if (x >= w.right() - 15) { close(w.app); return true; }
        if (x >= w.right() - 30) { toggleMaximize(w.app); return true; }
        if (x >= w.right() - 45) { minimize(w.app); return true; }
        long now = System.nanoTime();
        if (lastClickApp == w.app && now - lastClickAt < 350_000_000L) {
          toggleMaximize(w.app);
          lastClickAt = 0;
          return true;
        }
        lastClickAt = now;
        lastClickApp = w.app;
        dragging = w;
        moved = false;
        grabX = x - w.x; grabY = y - w.y;
        dragStartX = x; dragStartY = y;
        return true;
      }
      // A lower window consumes the first click to focus it, preventing click-through.
      return !wasFront;
    }
    return false;
  }

  public boolean pointerDragged(int x, int y) {
    if (resizing != null) {
      Window w = resizing;
      w.width = Math.max(210, Math.min(PixelCanvas.W - w.x, resizeWidth + x - dragStartX));
      w.height = Math.max(125, Math.min(BOTTOM - w.y, resizeHeight + y - dragStartY));
      w.restoreWidth = w.width; w.restoreHeight = w.height;
      return true;
    }
    if (dragging == null) return false;
    Window w = dragging;
    if (Math.abs(x - dragStartX) + Math.abs(y - dragStartY) > 3) moved = true;
    if (!moved) return true;
    if (w.maximized) {
      toggleMaximize(w.app);
      grabX = Math.min(w.width - 30, Math.max(18, grabX));
      grabY = 9;
    }
    w.x = x - grabX; w.y = y - grabY;
    clamp(w);
    return true;
  }

  public boolean pointerUp(int x, int y) {
    if (resizing != null) {
      resizing = null;
      return true;
    }
    if (dragging == null) return false;
    Window w = dragging;
    dragging = null;
    if (moved && y <= TOP + 4 && !w.maximized) toggleMaximize(w.app);
    else if (moved && x <= 3 && !w.maximized) snap(w, true);
    else if (moved && x >= PixelCanvas.W - 4 && !w.maximized) snap(w, false);
    if (moved) lastClickAt = 0; // Dragging must not count as a double-click.
    moved = false;
    return true;
  }

  public void cancelDrag() { dragging = null; resizing = null; moved = false; }

  private static void snap(Window w, boolean left) {
    w.restoreX = w.x; w.restoreY = w.y;
    w.restoreWidth = w.width; w.restoreHeight = w.height;
    w.x = left ? 2 : 242; w.y = TOP + 1;
    w.width = 236; w.height = BOTTOM - TOP - 2;
  }

  public void pointerMove(int x, int y) { hoverX = x; hoverY = y; }

  private static boolean inside(Window w, int x, int y) {
    return x >= w.x && x < w.right() && y >= w.y && y < w.bottom();
  }

  private static void clamp(Window w) {
    w.x = Math.max(0, Math.min(PixelCanvas.W - w.width, w.x));
    w.y = Math.max(TOP + 1, Math.min(BOTTOM - w.height, w.y));
  }

  public void tick(float delta) {
    time += delta;
    for (Window w : windows.values()) {
      float target = w.visible && !w.minimized ? 1 : 0;
      float speed = Math.min(1, delta * 13);
      w.animation += (target - w.animation) * speed;
      if (Math.abs(w.animation - target) < .008f) w.animation = target;
    }
    zOrder.removeIf(w -> !w.visible && w.animation == 0);
  }

  public void draw(PixelCanvas c, ContentPainter painter) {
    for (Window w : zOrder) {
      if (w.animation < .03f) continue;
      if (isMaximizedCode() && w.app != App.CODE) continue; // completely covered
      boolean active = front() == w;
      if (isMaximizedCode() && w.app == App.CODE) {
        // Legacy Workbench screens are drawn at the native 480x270 layout.
        painter.paint(w, true);
        drawCaption(c, w, true);
        continue;
      }
      float p = w.animation * w.animation * (3 - 2 * w.animation);
      int inset = Math.round((1 - p) * 13);
      int xx = w.x + inset, yy = w.y + inset;
      int ww = w.width - inset * 2, hh = w.height - inset * 2;
      if (ww < 50 || hh < 30) continue;
      c.rect(xx + 3, yy + 4, ww, hh, "020912a0");
      c.rect(xx, yy, ww, hh, active ? EDGE : "344a5e");
      c.rect(xx + 1, yy + 1, ww - 2, hh - 2, "0e1d2d");
      c.rect(xx + 1, yy + 1, ww - 2, 18, active ? "244359" : "1b2f41");
      if (w.animation > .94f) {
        c.clip(w.x + 2, w.y + 20, w.width - 4, w.height - 22);
        painter.paint(w, active && w.visible && !w.minimized);
        c.unclip();
        drawCaption(c, w, active);
      }
    }
  }

  private void drawCaption(PixelCanvas c, Window w, boolean active) {
    int y = w.y;
    c.rect(w.x, y, w.width, 18, active ? "25475e" : "1b3042");
    c.rect(w.x, y + 17, w.width, 1, active ? "5591a9" : "334a5c");
    c.fitted("◆  " + w.title, w.x + 8, y + 4, w.width - 65,
        active ? TEXT : "9eb3c0", true);
    int start = w.right() - 43;
    if (!w.maximized) {
      c.rect(w.right() - 7, w.bottom() - 4, 5, 1, "6695a8");
      c.rect(w.right() - 4, w.bottom() - 7, 1, 5, "6695a8");
    }
    for (int i = 0; i < 3; i++) {
      int x = start + i * 15;
      boolean hovered = hoverY >= y + 2 && hoverY <= y + 16 &&
          hoverX >= x && hoverX <= x + 12;
      c.rect(x, y + 3, 12, 12, hovered ? (i == 2 ? "b6525d" : "4a7c92")
          : i == 2 ? "704850" : "34576a");
      c.small(i == 0 ? "_" : i == 1 ? "□" : "×", x + 3, y + 2,
          i == 2 ? "ffd5d1" : ACCENT);
    }
  }

  public void reset() {
    zOrder.clear();
    dragging = null;
    resizing = null;
    for (Window w : windows.values()) {
      w.visible = false; w.minimized = false; w.maximized = false; w.animation = 0;
      // Preserve each window's restored size/position across visits to the in-game PC.
      w.x = w.restoreX; w.y = w.restoreY;
      w.width = w.restoreWidth; w.height = w.restoreHeight;
    }
  }

  public List<App> taskbar() {
    return zOrder.stream().filter(w -> w.visible).map(w -> w.app).toList();
  }

  public void cycleFocus() {
    if (zOrder.isEmpty()) return;
    List<Window> candidates = zOrder.stream().filter(w -> w.visible).toList();
    if (candidates.size() < 2) return;
    Window next = candidates.get(Math.max(0, candidates.size() - 2));
    focus(next.app);
  }
}

  private int operations(String source, LifeContent.Job j) {
    try {
      var p = ByteScript.compile(source);
      int max = 0;
      for (var t : j.tests) max = Math.max(max, p.operations(t.args));
      return max;
    } catch (RuntimeException e) {
      return -1;
    }
  }

          "hero", "hero_1", "hero_2", "mira", "arkady", "lev", "asya", "nika", "tamara"
        }) {
      Texture t = texture(n);
      int d = highDensity.contains(n) ? 2 : 1;
      actors.put(n, TextureRegion.split(t, 16 * d, 24 * d));
    }
  }

  public Texture texture(String name) {
    return textures.computeIfAbsent(
        name,
        n -> {
          var hd = Gdx.files.internal("art/hd/" + n + ".png");
          if (hd.exists()) highDensity.add(n);
          Texture t = new Texture(hd.exists() ? hd : Gdx.files.internal("art/" + n + ".png"));
          var filter =
              n.startsWith("light_") || n.equals("shadow")
                  ? Texture.TextureFilter.Linear
                  : Texture.TextureFilter.Nearest;
          t.setFilter(filter, filter);
          return t;
        });
  }

  public void begin() {
    int target =
        Math.max(
            1,
            Math.min(
                Gdx.graphics.getBackBufferWidth() / W, Gdx.graphics.getBackBufferHeight() / H));
    if (frame == null || target != density) {
      if (frame != null) frame.dispose();
      density = target;
      frame = new FrameBuffer(Pixmap.Format.RGBA8888, W * density, H * density, false);
      frame
          .getColorBufferTexture()
          .setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
    }
    clips.clear();
    frame.begin();
    Gdx.gl.glDisable(GL20.GL_SCISSOR_TEST);
    Gdx.gl.glClearColor(.075f, .1f, .14f, 1);
    Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);
    batch.setProjectionMatrix(camera.combined);
    batch.begin();
    batch.setColor(Color.WHITE);
  }

  public void end() {
    if (!clips.isEmpty()) throw new IllegalStateException("Unbalanced render clips");
    batch.end();
    frame.end();
    int w = Gdx.graphics.getWidth(), h = Gdx.graphics.getHeight();
    scale = Math.max(1, Math.min(w / W, h / H));
    offsetX = (w - W * scale) / 2;
    offsetY = (h - H * scale) / 2;
    Gdx.gl.glViewport(0, 0, Gdx.graphics.getBackBufferWidth(), Gdx.graphics.getBackBufferHeight());
    Gdx.gl.glClearColor(.045f, .06f, .085f, 1);
    Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);
    camera.setToOrtho(false, w, h);
    batch.setProjectionMatrix(camera.combined);
    batch.begin();
    batch.setColor(Color.WHITE);
    batch.draw(frame.getColorBufferTexture(), offsetX, offsetY, W * scale, H * scale, 0, 0, 1, 1);
    batch.end();
    camera.setToOrtho(false, W, H);
  }

  public float artWidth(String name) {
    Texture t = texture(name);
    return t.getWidth() / (highDensity.contains(name) ? 2f : 1f);
  }

  public float artHeight(String name) {
    Texture t = texture(name);
    return t.getHeight() / (highDensity.contains(name) ? 2f : 1f);
  }

  /** Nested framebuffer scissor; every layer is intersected with its parent. */
  public void clip(float x, float y, float w, float h) {
    batch.flush();
    float x0 = Math.max(0, x),
        y0 = Math.max(0, y),
        x1 = Math.min(W, x + w),
        y1 = Math.min(H, y + h);
    if (!clips.isEmpty()) {
      var p = clips.peek();
      x0 = Math.max(x0, p.x);
      y0 = Math.max(y0, p.y);
      x1 = Math.min(x1, p.x + p.width);
      y1 = Math.min(y1, p.y + p.height);
    }
    clips.push(
        new com.badlogic.gdx.math.Rectangle(x0, y0, Math.max(0, x1 - x0), Math.max(0, y1 - y0)));
    applyClip();
  }

  private void applyClip() {
    if (clips.isEmpty()) {
      Gdx.gl.glDisable(GL20.GL_SCISSOR_TEST);
      return;
    }
    var p = clips.peek();
    int x0 = (int) Math.ceil(p.x * density), y0 = (int) Math.ceil(p.y * density);
    int x1 = (int) Math.floor((p.x + p.width) * density),
        y1 = (int) Math.floor((p.y + p.height) * density);
    Gdx.gl.glEnable(GL20.GL_SCISSOR_TEST);
    Gdx.gl.glScissor(x0, H * density - y1, Math.max(0, x1 - x0), Math.max(0, y1 - y0));
  }

  public void unclip() {
    batch.flush();
    if (clips.isEmpty()) throw new IllegalStateException("Missing render clip");
    clips.pop();
    applyClip();
  }

  public void fitted(String text, float x, float y, float width, String color, boolean compact) {
    BitmapFont f = compact ? small : font;
    layout.setText(f, text);
    if (layout.width > width) {
      int end = text.length();
      do {
        end--;
        layout.setText(f, text.substring(0, end) + "…");
      } while (end > 0 && layout.width > width);
      text = text.substring(0, end) + "…";
    }
    clip(x, y - 1, width, compact ? 13 : 16);
    if (compact) small(text, x, y, color);
    else text(text, x, y, color);
    unclip();
  }

  public void glow(String name, float x, float y, float strength) {
    batch.setColor(1, 1, 1, Math.max(0, Math.min(1, strength)));
    art(name, x, y);
    batch.setColor(Color.WHITE);
  }

  public void rect(float x, float y, float w, float h, Color color) {
    batch.setColor(color);
    batch.draw(white, x, H - y - h, w, h);
    batch.setColor(Color.WHITE);
  }

  public void rect(float x, float y, float w, float h, String color) {
    rect(x, y, w, h, color(color));
  }

  /**
   * Smooth high-density rounded panels. A cached three-times-resolution signed-
   * distance mask avoids the chunky stepped corners of the low-res prototype.
   * The mask is white and can be tinted to any colour by the SpriteBatch.
   */
  public void roundRect(float x, float y, float w, float h, int radius, String hex) {
    int r = Math.max(0, Math.min(radius, (int)Math.min(w, h) / 2));
    if (r == 0 || w < 1 || h < 1) { rect(x, y, w, h, hex); return; }
    int pw = Math.max(2, Math.round(w * 3)), ph = Math.max(2, Math.round(h * 3));
    String key = pw + ":" + ph + ":" + r;
    Texture mask = roundedMasks.get(key);
    if (mask == null) {
      Pixmap p = new Pixmap(pw, ph, Pixmap.Format.RGBA8888);
      float radiusPx = r * 3f;
      float bx = pw / 2f - radiusPx;
      float by = ph / 2f - radiusPx;
      for (int py = 0; py < ph; py++) {
        float qy = Math.abs(py + .5f - ph / 2f) - by;
        for (int px = 0; px < pw; px++) {
          float qx = Math.abs(px + .5f - pw / 2f) - bx;
          float ax = Math.max(0, qx), ay = Math.max(0, qy);
          float distance = (float)Math.sqrt(ax * ax + ay * ay)
              + Math.min(0, Math.max(qx, qy)) - radiusPx;
          int alpha = Math.round(Math.max(0, Math.min(1, .5f - distance)) * 255);
          p.drawPixel(px, py, 0xffffff00 | alpha);
        }
      }
      mask = new Texture(p);
      mask.setFilter(Texture.TextureFilter.Linear, Texture.TextureFilter.Linear);
      p.dispose();
      roundedMasks.put(key, mask);
    }
    batch.setColor(color(hex));
    batch.draw(mask, x, H - y - h, w, h);
    batch.setColor(Color.WHITE);
  }

  public void panel(int x, int y, int w, int h) {
    rect(x + 1, y + 2, w, h, "07141cc0");
    rect(x, y, w, h, "526b70");
    rect(x + .5f, y + .5f, w - 1, h - 1, "172e3bf9");
    rect(x + 1, y + 1, w - 2, .5f, "829b9160");
  }

  public void art(String name, float x, float y) {
    Texture t = texture(name);
    batch.draw(
        t, Math.round(x), H - Math.round(y) - artHeight(name), artWidth(name), artHeight(name));
  }

  public void scaledArt(String name, float x, float y, float width, float height) {
    batch.draw(texture(name), Math.round(x), H - Math.round(y) - height, width, height);
  }

  public void actor(String name, int dir, int animation, float x, float feetY, int outfit) {
    if (name.equals("hero") && outfit > 0) name = "hero_" + outfit;
    batch.draw(actors.get(name)[dir][animation], Math.round(x) - 8, H - Math.round(feetY), 16, 24);
    batch.setColor(Color.WHITE);
  }

  public void text(String text, float x, float y, Color color) {
    font.setColor(color);
    font.draw(batch, text, Math.round(x), H - Math.round(y));
  }

  public void text(String text, float x, float y, String color) {
    text(text, x, y, color(color));
  }

  public void small(String text, float x, float y, String color) {
    small.setColor(color(color));
    small.draw(batch, text, Math.round(x), H - Math.round(y));
  }

  public void title(String text, float x, float y, String color) {
    title.setColor(color(color));
    title.draw(batch, text, Math.round(x), H - Math.round(y));
  }

  public void wrap(String text, float x, float y, float width, String color) {
    wrap(text, x, y, width, H - y - 8, color);
  }

  public void wrap(String text, float x, float y, float width, float height, String color) {
    clip(x, y - 1, Math.min(width, W - x - 12), height);
    font.setColor(color(color));
    font.draw(batch, text, x, H - y, width, com.badlogic.gdx.utils.Align.left, true);
    unclip();
  }

  public float width(String text) {
    layout.setText(font, text);
    return layout.width;
  }

  public void center(String text, float y, String color) {
    text(text, Math.round((W - width(text)) / 2), y, color);
  }

  public float logicalX(int x) {
    return (x - offsetX) / (float) scale;
  }

  public float logicalY(int y) {
    return (y - offsetY) / (float) scale;
  }

  public float mouseX() {
    return logicalX(Gdx.input.getX());
  }

  public float mouseY() {
    return logicalY(Gdx.input.getY());
  }

  public void screenshot(String path) {
    Pixmap pix =
        Pixmap.createFromFrameBuffer(
            0, 0, Gdx.graphics.getBackBufferWidth(), Gdx.graphics.getBackBufferHeight());
    PixmapIO.writePNG(Gdx.files.absolute(path), pix, -1, true);
    pix.dispose();
  }

  public void dispose() {
    font.dispose();
    small.dispose();
    title.dispose();
    textures.values().forEach(Texture::dispose);
    roundedMasks.values().forEach(Texture::dispose);
    white.dispose();
    if (frame != null) frame.dispose();
    batch.dispose();
  }
}

# Pinux Astra: Archcraft-quality UI rebuild (decision record)

Status: **NOT APPROVED / PROTOTYPE ONLY**. The existence of passing Java or smoke tests does not mean the interface is visually acceptable. This document is a quality contract for the \`ctrl-life\` branch, not a claim that the design is finished.

## Evidence: failures of the previous UI

Review the real screenshots saved by \`.github/workflows/pinux-desktop-preview.yml\` rather than concept art.

1. The 480x270 \`PixelCanvas\` workbench becomes a **three-times enlarged** older-game UI inside a crisp desktop. The hybrid must end. All Pinux applications, including jobs, ByteScript and Astra Code, need to render through one resolution-independent UI layer.
2. A library such as VisUI solves pointer handling, drag and resize, **not** visual design. Unmodified colours, procedural symbols and generic labels do not resemble Archcraft.
3. The generated city-and-sunset wallpaper dominates the composition and reads as a cartoon. Professional Archcraft "rices" use restrained, carefully composed background art.
4. Earlier desktop shortcuts were 120x113, the topbar was 52 high, the dock 415x80; Archcraft Openbox's Polybar can be as compact as 26 high. Those proportions were wrong.
5. Tiny bitmap glyphs scaled at runtime look soft; the terminal had no distinctive monospace typeface. Rendering in a native-resolution \`Stage\` is necessary but not sufficient.

## Technical decision

**Keep Java 21 and libGDX for the native game. Keep Scene2D's Stage and input handling for the OS. Replace the entire Pinux presentation, including the legacy workbench, with one coherent native-resolution design system.**

Do not migrate to Java Swing, JavaFX, Compose Desktop or a separate browser window. All of those create an additional window/runtime or require fragile off-screen compositing. Dear ImGui Java supports docking and makes sense for an editor, but its programmer-tool defaults still need a full design system and make the OS shell harder to style. A blind framework swap would repeat the same failure.

A future Dear ImGui proof-of-concept is acceptable only **inside the existing LWJGL3 window**, and only if it beats Scene2D in verified code-editing behavior and visual quality. It is not an excuse to ship default ImGui styling.

## Archcraft reference, precise design sources

- https://wiki.archcraft.io/docs/window-managers/stacking-wm/openbox/
- https://github.com/archcraft-os/archcraft-openbox/tree/main/files/themes
- https://github.com/archcraft-os/archcraft-bspwm/tree/main/files/themes/onedark
- \`files/themes/onedark/polybar/colors.ini\` and \`rofi/shared/colors.rasi\` define OneDark's actual palette.
- \`files/themes/*/polybar/config.ini\` gives panel density and hierarchy.
- \`files/themes/*/rofi/launcher.rasi\` gives an actionable launcher composition.

Archcraft's upstream package repos include GPLv3 content. **Use screenshots and style measurements as reference, do not copy GPL implementation/assets into the existing MIT game.** Keep notices for every imported asset with a compatible licence.

## Design tokens (first pass, subject to visual verification)

| Token | Value |
| --- | --- |
| OS background | #1E2128 |
| Raised surface | #292D37 |
| Chrome/window title | #292D37 |

Use libGDX FreeType to generate properly hinted font bitmaps for each pixel size; avoid stretching one texture atlas for multiple UI sizes. If using SDF fonts, test small-size legibility. Convert selected SVG icons to antialiased PNG atlases **at build time**; bundle them with the game, never require network access at runtime.

## Required architecture

\`\`\`
CtrlLife (libGDX native window)
├─ PixelCanvas — world, movement, sprites and scenery only
└─ PinuxStage — all PC UI at window pixel resolution
   ├─ Wallpaper + theme
   ├─ Polybar (workspaces, active window, system/game indicators)
   ├─ Rofi-style launcher (real search/filter/keyboard selection)
   ├─ WindowManager (drag, resize, max/min/restore, focus, snap)
   ├─ AstraCodeView — Explorer, tabs, editor, tests, diagnostics
   ├─ TerminalView — terminal model, history, input, scroll
   ├─ FilesView — game virtual file system
   ├─ MailView — real existing career/job offers
   └─ SettingsView — working theme and accessibility controls

ComputerViewModel — adapted from existing Career/Coding/GameState
VirtualFS and TerminalModel — **game sandbox only**
\`\`\`

No UI markup or model for the PC should directly call host shell/filesystem operations. Preserve ByteScript and gameplay logic. The biggest migration is to replace \`workbench.draw()\` on the low-res canvas with a native-resolution AstraCodeView, while preserving the existing game-state controller.

## Milestones, one at a time

1. **Design system + shell:** consistent fonts, icons, wallpaper, 30 px Polybar, 40 px dock, keyboard launcher. Remove oversized desktop cards. Screenshot desktop and launcher.
2. **Window chrome:** consistent titlebar, correct focus/drag/resize/max/min, scale at 720p/810p/1080p; verify title text and content do not clip.
3. **Terminal:** actual monospace font, selection and cursor, history, copy/paste within game sandbox, scrolling, run \`help\` and \`ls\`.
4. **Astra Code:** remove old 480x270 workbench UI. Reimplement functional project panel/editor/tests in Scene2D with underlying ByteScript logic unchanged.
5. **Files, TaskMail, Settings:** each uses shared components and real game models, not placeholder content. Full navigation/input tests.
6. **Packaging:** build complete Windows x64 ZIP including Temurin 21 runtime and licensed assets; clean machine launch without installed Java; link directly to playable build.

### Mandatory acceptance gates

For EVERY stage:
- Run real \`:core:test\` and \`:lwjgl3:guiSmoke\` on Xvfb.
- Save and **inspect** actual full-resolution PNG screenshots: desktop, focused window, other window, resizing, launcher. Never call an image-generation concept a screenshot.
- Check 1280x720, 1440x810, and 1920x1080 for clipping, legibility and collisions. Include visible hover, active, disabled and focus cases.
- Test mouse input, keyboard focus, ESC, minimize/restore, app open/close, resize, and Alt+Tab (or documented equivalent).
- Address visible defects and rerun before considering the stage complete.
- No release until the full Windows ZIP has been built, tested and verified to contain \`runtime/bin/java.exe\`, \`run.bat\` and all game jars.

Passing automated tests means "functional smoke passed", **never** "design approved". Visual approval requires reviewing screenshots.

It proves the underlying Scene2D architecture can support realistic desktop dimensions **without rewriting the game engine**, but this is not a final UI: the wallpaper/art, terminal typography and full native Astra Code migration are still unresolved.

# Pinux Astra — Global Update 0.5.0

Это вымышленная ОС **внутри CTRL+LIFE**, а не веб-страница, виртуальная машина и не доступ к компьютеру игрока. Реализована на Java/libGDX: игровой мир по-прежнему рисуется через \`PixelCanvas\`, а приложения и рабочий стол используют Scene2D/VisUI при настоящем разрешении экрана; терминал изолирован от host OS.

## Native Linux-style UI foundation (VisUI)

The desktop and newly ported applications now render at the **actual game-window resolution**, not the 480×270 pixel-art framebuffer.
The window layer uses [VisUI](https://github.com/kotcrab/vis-ui) 1.5.6 (Apache-2.0; matches libGDX 1.13.1), including `VisWindow` for drag, resize and stacking and `VisTextField` for the in-game terminal.
The architecture follows the separation found in [gezhov/Fake-OS-Desktop](https://github.com/gezhov/Fake-OS-Desktop) (MIT): shell, independent app windows, workspace, taskbar and focus. Swing classes were not copied because they are incompatible with libGDX's render loop.

Native screens: **Astra Code**, Terminal, Files, TaskMail and Settings. Astra Code полностью переведён в отдельный native-resolution VisUI window. Его экраны больше не зависят от старой 480×270 разметки. Реальные игровые результаты продолжают рассчитываться классами \`Coding\`, \`Career\` и \`ByteScript\`, а состояние — \`GameState\`. Terminal commands are **game simulations only** and never execute real commands.

**Visual quality gate:** `pinux-desktop-preview.yml` runs an actual GLFW/libGDX smoke under Xvfb, captures PNGs, and uploads them as a workflow artifact. We review new captures before moving to the next screen.

## Window manager

Отдельный `PinuxWindowManager` отвечает за z-order, фокус, двойное нажатие на заголовок, анимации открытия/сворачивания/закрытия, минимизацию, восстановление, перетаскивание, изменение размеров окна за правый нижний угол, прижатие к левому/правому краю, максимизацию при перетаскивании к верхнему краю, расположение окон на рабочем столе. Клики по заголовкам перехватываются до глобальных кнопок, чтобы предотвратить click-through. Клавиша Tab циклически меняет активное окно.

## Что уже есть

- Заставка загрузки, детализированный рабочий стол Pinux Astra, верхняя системная панель с внутриигровым временем, меню приложений и нижний dock.
- Приложения: **Astra Code**, **Terminal**, **Files**, **TaskMail**, **Settings**. Есть синяя и тёплая темы, режим уменьшения анимаций.
- Astra Code использует знакомую композицию IDE: activity rail → Explorer → вкладка \`main.bs\` → исходный код → панель проверок → действия.
- Терминал открывается как отдельное перемещаемое окно поверх рабочего места Astra Code; клавиатурный ввод переключается в активное окно консоли, а не в автопечать кода.
- Файловый экран показывает игровую структуру проекта, не реальные диски.
- Новая фаза **набора черновика**: в заданиях IDE любое обычное нажатие клавиши добавляет символы в подготовленный исходник; видимый код появляется постепенно.
- После проваленных тестов **следующее обычное нажатие клавиши автоматически запускает фазу исправления**: исправленный код появляется через ту же механику набора. Ручной выбор варианта патча в старом интерфейсе сохранён как дополнительный инструмент. Повторный прогон тестов нужен после завершения ввода.
- Используется **настоящая проверка ByteScript** из прежней версии игры: тесты, синтаксические ошибки, операции и локальная история версий сохраняются.

- \`life.ctrl.ui.AstraCodeApp\`: полнофункциональный интерфейс высокого разрешения: контракты, ByteScript-редактор, тесты, отладка, предпросмотр, лаборатория, Git, карьера, железо, портфолио, собственный бот.

- \`life.ctrl.ui.LifeScreens\`: связь игрового ПК с оболочкой.
- \`life.ctrl.CtrlLife\`: передача нажатий в оболочку при режиме \`IDE\`.
- \`life.ctrl.model.Coding\`: перегрузки \`accept(..., tactile)\` и \`repair(..., tactile)\`. Отдельный выбор режима защищает прямые доменные тесты от несовместимых изменений.

## Ограничения Global Update

Astra Code — полноценное масштабируемое окно VisUI (перемещение, изменение размера, максимизация и сворачивание). Весь цикл заказов от выбора клиента до выплаты проходит внутри native-resolution интерфейса. ByteScript компилирует и выполняет настоящий игровой исходник; платёж, опыт, прогресс, бонусы и версии сохраняются в игровом состоянии. Нажатия мышью обрабатывает Scene2D в координатах реального окна; пиксельные координаты оставлены для игрового мира. Терминал не запускает программы ПК игрока.
